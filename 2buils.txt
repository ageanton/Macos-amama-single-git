              private enum MarkdownBlockKind {

                  case heading(Int, String)

              private func sanitizeIncompleteStreamMarkers(_ input: String, isStreaming: Bool) -> String {
                  guard isStreaming else { return input }
                  var result = input
                  let boldCount = result.components(separatedBy: "**").count - 1
                  if boldCount % 2 != 0, let lastRange = result.range(of: "**", options: .backwards) {
                      result.removeSubrange(lastRange)
                  }
                  let fenceCount = result.components(separatedBy: "```").count - 1
                  if fenceCount % 2 == 0 {
                      let inlineBackticks = result.components(separatedBy: "`").count - 1
                      if inlineBackticks % 2 != 0, let lastTick = result.range(of: "`", options: .backwards) {
                          result.removeSubrange(lastTick)
                      }
                  }
                  return result
              }

              private func parseBlocks(from raw: String) -> [MarkdownBlock] {
                  var blocks: [MarkdownBlock] = []
                  let segments = raw.components(separatedBy: "```")

                  for (segIdx, segment) in segments.enumerated() {
                      if segIdx % 2 == 1 {
                          var lines = segment.components(separatedBy: .newlines)
                          let firstLine = lines.first?.trimmingCharacters(in: .whitespacesAndNewlines) ?? ""
                          if !lines.isEmpty { lines.removeFirst() }
                          let codeBody = lines.joined(separator: "\n")
                          blocks.append(MarkdownBlock(kind: .codeBlock(firstLine, codeBody)))
                      } else {
                          let lines = segment.components(separatedBy: .newlines)
                          var currentParagraph: [String] = []

                          func flushParagraph() {
                              let joined = currentParagraph.joined(separator: "\n").trimmingCharacters(in: .whitespacesAndNewlines)
                              if !joined.isEmpty {
                                  blocks.append(MarkdownBlock(kind: .paragraph(joined)))
                              }
                              currentParagraph.removeAll()
                          }

                          for line in lines {
                              let trimmed = line.trimmingCharacters(in: .whitespaces)
                              if trimmed.isEmpty {
                                  flushParagraph()
                              } else if trimmed == "---" || trimmed == "***" {
                                  flushParagraph()
                                  blocks.append(MarkdownBlock(kind: .horizontalRule))
                              } else if trimmed.hasPrefix("### ") {
                                  flushParagraph()
                                  blocks.append(MarkdownBlock(kind: .heading(3, String(trimmed.dropFirst(4)))))
                              } else if trimmed.hasPrefix("## ") {
                                  flushParagraph()
                                  blocks.append(MarkdownBlock(kind: .heading(2, String(trimmed.dropFirst(3)))))
                              } else if trimmed.hasPrefix("# ") {
                                  flushParagraph()
                                  blocks.append(MarkdownBlock(kind: .heading(1, String(trimmed.dropFirst(2)))))
                              } else if trimmed.hasPrefix("- ") || trimmed.hasPrefix("* ") || trimmed.hasPrefix("• ") {
                                  flushParagraph()
                                  blocks.append(MarkdownBlock(kind: .bulletItem(String(trimmed.dropFirst(2)))))
                              } else if trimmed.hasPrefix("> ") {
                                  flushParagraph()
                                  blocks.append(MarkdownBlock(kind: .blockquote(String(trimmed.dropFirst(2)))))
                              } else if let numberedMatch = matchNumberedList(trimmed) {
                                  flushParagraph()
                                  blocks.append(MarkdownBlock(kind: .numberedItem(numberedMatch.number, numberedMatch.rest)))
                              } else {
                                  currentParagraph.append(line)
                              }
                          }
                          flushParagraph()
                      }
                  }
                  return blocks
              }

              private func matchNumberedList(_ line: String) -> (number: String, rest: String)? {
                  guard let dotIdx = line.firstIndex(of: ".") else { return nil }
                  let prefix = String(line[..<dotIdx])
                  guard !prefix.isEmpty, prefix.allSatisfy({ $0.isNumber }) else { return nil }
                  let afterDot = line.index(after: dotIdx)
                  guard afterDot < line.endIndex, line[afterDot] == " " else { return nil }
                  let body = String(line[line.index(after: afterDot)...])
                  return ("\(prefix).", body)
              }

                  var parent: IMESafeComposerTextView
                  init(_ parent: IMESafeComposerTextView) { self.parent = parent }

                  public func textDidBeginEditing(_ notification: Notification) { parent.isFocused = true }
                  public func textDidEndEditing(_ notification: Notification) { parent.isFocused = false }

                  public func textDidChange(_ notification: Notification) {
                      guard let tv = notification.object as? NSTextView else { return }
                      if !tv.hasMarkedText() {
                          parent.text = tv.string

                  /// Uses AppKit's NSLayoutManager to compute the exact pixel height of wrapped lines (whether wrapped naturally or via Shift+Return)
                  public func recalculateWrappedHeight(for textView: NSTextView) {
                      guard let layoutManager = textView.layoutManager,
                            let textContainer = textView.textContainer else { return }
                      layoutManager.ensureLayout(for: textContainer)
                      let usedHeight = layoutManager.usedRect(for: textContainer).height
                      let insetHeight = textView.textContainerInset.height * 2.0
                      let rawHeight = textView.string.isEmpty ? 32.0 : ceil(usedHeight + insetHeight + 2.0)
                      let clamped = min(148.0, max(32.0, rawHeight))
                      if abs(parent.measuredHeight - clamped) > 0.5 {
                          DispatchQueue.main.async {
                              self.parent.measuredHeight = clamped
                          }
                      }
                  }

                  public func textView(_ textView: NSTextView, doCommandBy commandSelector: Selector) -> Bool {
                      if textView.hasMarkedText() { return false }

                              parent.onSlashMoveUp()
                              return true
                          }
                          if commandSelector == #selector(NSResponder.moveDown(_:)) {
                              parent.onSlashMoveDown()
                              return true
                          }
                          if commandSelector == #selector(NSResponder.insertTab(_:)) {
                              _ = parent.onSlashConfirm()
                              return true
                          }
                          if commandSelector == #selector(NSResponder.cancelOperation(_:)) {
                              parent.onSlashDismiss()
                              return true
                          }
                      }

                      if commandSelector == #selector(NSResponder.insertNewline(_:)) {
                          let flags = NSApp.currentEvent?.modifierFlags ?? []
                          if flags.contains(.shift) {
                              textView.insertNewlineIgnoringFieldEditor(nil)
                              recalculateWrappedHeight(for: textView)
                              return true
                          } else {

                                  recalculateWrappedHeight(for: textView)
                                  return true
                              }
                              parent.onSend()
                              return true
                          }
                      }
                      return false
                  }
              }

              private func saveArtifactToDisk(_ artifact: WorkspaceArtifact) {
                  let panel = NSSavePanel()
                  panel.title = "Save Code Snippet"
                  let ext = artifact.language.isEmpty ? "txt" : artifact.language
                  panel.nameFieldStringValue = "\(artifact.title.replacingOccurrences(of: " ", with: "_")).\(ext)"
                  if panel.runModal() == .OK, let url = panel.url {
                      try? artifact.content.write(to: url, atomically: true, encoding: .utf8)
                  }
              }

              private func openArtifactInExternalEditor(_ artifact: WorkspaceArtifact) {
                  let ext = artifact.language.isEmpty ? "txt" : artifact.language
                  let tempURL = FileManager.default.temporaryDirectory.appendingPathComponent("MotifSnippet.\(ext)")
                  try? artifact.content.write(to: tempURL, atomically: true, encoding: .utf8)
                  NSWorkspace.shared.open(tempURL)
              }

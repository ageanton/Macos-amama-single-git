    /// Peripheral-vision "Model Ready" choreography: emerald Horizon Sweep + Composer Halo + 1.9s Ready Pill.
    public func triggerModelLoadedCelebration(model: LocalModelDescriptor, ctx: Int) {
        triggerModelLoadCompletionFlicker()
        let gpuText: String
        if let frac = model.gpuOffloadFraction {
            let pct = Int(round(frac * 100.0))
            gpuText = pct >= 98 ? "100% GPU" : "\(pct)% GPU"
        } else {
            gpuText = "In RAM"
        }
        let loadedCtx = model.loadedContextLength ?? ctx
        let banner = "\(model.shortDisplayName) ready · \(gpuText) · \(loadedCtx / 1024)k ctx"
        let myPulse = UUID()
        modelReadySweepID = myPulse

        Task { @MainActor in
            withAnimation(.spring(response: 0.32, dampingFraction: 0.82)) {
                self.modelReadyBannerText = banner
                self.composerHaloOpacity = 1.0
            }
            try? await Task.sleep(nanoseconds: 1_100_000_000)
            withAnimation(.easeOut(duration: 0.45)) {
                self.composerHaloOpacity = 0.0
            }
            try? await Task.sleep(nanoseconds: 850_000_000)
            if self.modelReadySweepID == myPulse {
                withAnimation(.easeOut(duration: 0.30)) {
                    self.modelReadyBannerText = nil
                }
            }
        }
    }

        }

        public func recalculateQuantizedMetrics(for textView: NSTextView) {
            if textView.string.isEmpty {
                if parent.lineCount != 1 || parent.compactHeight != 28.0 {
                    DispatchQueue.main.async {
                        self.parent.lineCount = 1
                        self.parent.compactHeight = 28.0
                    }
                }
                return
            }
            guard let lm = textView.layoutManager, let tc = textView.textContainer else { return }
            lm.ensureLayout(for: tc)
            let usedH = lm.usedRect(for: tc).height
            let singleH = max(18.0, (textView.font?.ascender ?? 13.0) - (textView.font?.descender ?? -4.0) + (textView.font?.leading ?? 2.0))
            let lines = max(1, Int(round(usedH / singleH)))
            let targetH: CGFloat = lines <= 1 ? 28.0 : (lines == 2 ? 46.0 : 64.0)
            if parent.lineCount != lines || abs(parent.compactHeight - targetH) > 0.5 {
                DispatchQueue.main.async {
                    self.parent.lineCount = lines
                    self.parent.compactHeight = targetH
                }
            }
        }

        public func textView(_ textView: NSTextView, doCommandBy commandSelector: Selector) -> Bool {
            if textView.hasMarkedText() { return false }
            if parent.isSlashMenuVisible {
                if commandSelector == #selector(NSResponder.moveUp(_:)) { parent.onSlashMoveUp(); return true }
                if commandSelector == #selector(NSResponder.moveDown(_:)) { parent.onSlashMoveDown(); return true }
                if commandSelector == #selector(NSResponder.insertTab(_:)) {
                    let ok = parent.onSlashConfirm()
                    recalculateQuantizedMetrics(for: textView)
                    return ok
                }
                if commandSelector == #selector(NSResponder.cancelOperation(_:)) { parent.onSlashDismiss(); return true }
            }
            if commandSelector == #selector(NSResponder.insertNewline(_:)) {
                let flags = NSApp.currentEvent?.modifierFlags ?? []
                if flags.contains(.shift) {
                    textView.insertNewlineIgnoringFieldEditor(nil)
                    recalculateQuantizedMetrics(for: textView)
                    return true
                } else {
                    if parent.isSlashMenuVisible && parent.onSlashConfirm() {
                        recalculateQuantizedMetrics(for: textView)
                        return true
                    }
                    parent.onSend()
                    return true
                }
            }
            return false
        }
    }
}

    private func handleDroppedProviders(_ providers: [NSItemProvider]) -> Bool {
        guard store.activeSession != nil else { return false }
        var handled = false
        for provider in providers {
            if provider.hasItemConformingToTypeIdentifier(UTType.fileURL.identifier) {
                handled = true
                provider.loadItem(forTypeIdentifier: UTType.fileURL.identifier, options: nil) { item, _ in
                    var resolvedURL: URL? = nil
                    if let data = item as? Data, let str = String(data: data, encoding: .utf8) {
                        resolvedURL = URL(string: str)
                    } else if let url = item as? URL {
                        resolvedURL = url
                    }
                    if let url = resolvedURL {
                        Task { @MainActor in store.attachFilesFromURLs([url]) }
                    }
                }
            } else if provider.hasItemConformingToTypeIdentifier(UTType.image.identifier) {
                handled = true
                provider.loadDataRepresentation(forTypeIdentifier: UTType.image.identifier) { data, _ in
                    if let data = data {
                        Task { @MainActor in store.handleClipboardImagePaste(data) }
                    }
                }
            }
        }
        return handled
    }

    // MARK: - Conversation Stream Canvas
    private func conversationStreamCanvas(
        session: ConversationSession,
        useLeftGutter: Bool,
        totalRowMaxWidth: CGFloat,
        horizontalPadding: CGFloat,
        opticalFontSize: CGFloat,
        isFullscreenOrWide: Bool
    ) -> some View {
        let thread = session.activeBranchThread
        let lastTurnID = thread.last?.id
        let dynamicBottomClearance = max(130.0, composerMeasuredHeight + 28.0)
        let jumpButtonBottomOffset = max(96.0, composerMeasuredHeight + 12.0)

        return ScrollViewReader { proxy in
            ScrollView(.vertical, showsIndicators: true) {
                LazyVStack(alignment: .leading, spacing: 26) {
                    ForEach(thread) { node in
                        messageTurnView(
                            node: node,
                            session: session,
                            isLatestTurn: node.id == lastTurnID,
                            useLeftGutter: useLeftGutter,
                            opticalFontSize: opticalFontSize,
                            isFullscreenOrWide: isFullscreenOrWide
                        )
                        .id(node.id)
                    }

                    Color.clear
                        .frame(height: dynamicBottomClearance)
                        .id("STREAM_BOTTOM_ANCHOR")
                }
                .background(
                    ScrollViewBottomObserver(
                        isScrolledAwayFromBottom: $store.isScrolledAwayFromBottom,
                        isDark: tokens.isEffectivelyDark
                    ).frame(width: 1, height: 1)
                )
                .frame(maxWidth: totalRowMaxWidth, alignment: .leading)
                .padding(.horizontal, horizontalPadding)
                .padding(.top, 22)
                .frame(maxWidth: .infinity)
            }
            .onChange(of: store.streamScrollPulse) { _, _ in
                if store.isStreamingGeneration && !store.isScrolledAwayFromBottom && !store.isAutoScrollPaused {
                    proxy.scrollTo("STREAM_BOTTOM_ANCHOR", anchor: .bottom)
                }
            }
            .onChange(of: store.pendingScrollToMessageID) { _, targetID in
                if let id = targetID {
                    withAnimation(.spring(response: 0.28, dampingFraction: 0.86)) { proxy.scrollTo(id, anchor: .center) }
                    store.pendingScrollToMessageID = nil
                }
            }
            .overlay(alignment: .bottom) {
                if store.isScrolledAwayFromBottom && !store.isComposerCanvasExpanded && store.matchedSlashPersonas.isEmpty {
                    HighContrastJumpToLatestButton(tokens: tokens) {
                        store.isScrolledAwayFromBottom = false
                        store.isAutoScrollPaused = false
                        withAnimation(.easeOut(duration: 0.18)) { proxy.scrollTo("STREAM_BOTTOM_ANCHOR", anchor: .bottom) }
                    }
                    .padding(.bottom, jumpButtonBottomOffset)
                    .transition(.scale.combined(with: .opacity))
                }
            }
        }
    }

    @ViewBuilder
    private func messageTurnView(
        node: ConversationMessageNode,
        session: ConversationSession,
        isLatestTurn: Bool,
        useLeftGutter: Bool,
        opticalFontSize: CGFloat,
        isFullscreenOrWide: Bool
    ) -> some View {
        let siblingMeta = session.siblingInfo(for: node.id)
        let isNodeStreaming = store.isStreamingGeneration && store.streamingNodeID == node.id

        VStack(alignment: .leading, spacing: 12) {
            if useLeftGutter {
                HStack(alignment: .top, spacing: 20) {
                    VStack(alignment: .leading, spacing: 7) {
                        turnRoleOrHotSwapMenu(node: node, maxLabelWidth: 122)
                        if siblingMeta.totalCount > 1 { branchPaginationControl(nodeID: node.id, siblingMeta: siblingMeta) }
                        if isNodeStreaming {
                            Button("Stop") { store.stopActiveGeneration() }
                                .font(typo.sans(size: 11, weight: .semibold))
                                .foregroundColor(tokens.accentAction)
                                .buttonStyle(.plain)
                        }
                    }
                    .frame(width: 128, alignment: .leading)
                    .clipped()

                    VStack(alignment: .leading, spacing: 10) {
                        turnMainBody(node: node, isNodeStreaming: isNodeStreaming, opticalFontSize: opticalFontSize, isFullscreenOrWide: isFullscreenOrWide)
                        turnBottomActionBar(node: node, session: session, isLatestTurn: isLatestTurn, isNodeStreaming: isNodeStreaming)
                    }
                    .frame(maxWidth: .infinity, alignment: .leading)
                }
            } else {
                HStack(alignment: .center, spacing: 8) {
                    turnRoleOrHotSwapMenu(node: node, maxLabelWidth: 210)
                    if siblingMeta.totalCount > 1 { branchPaginationControl(nodeID: node.id, siblingMeta: siblingMeta) }
                    Spacer()
                    if isNodeStreaming {
                        Button("Stop") { store.stopActiveGeneration() }
                            .font(typo.sans(size: 11.5, weight: .semibold))
                            .foregroundColor(tokens.accentAction)
                            .buttonStyle(.plain)
                    }
                }
                turnMainBody(node: node, isNodeStreaming: isNodeStreaming, opticalFontSize: opticalFontSize, isFullscreenOrWide: isFullscreenOrWide)
                turnBottomActionBar(node: node, session: session, isLatestTurn: isLatestTurn, isNodeStreaming: isNodeStreaming)
            }
            HairlineDivider(tokens: tokens).padding(.top, 6)
        }
        .contextMenu {
            Button("Copy Message") {
                NSPasteboard.general.clearContents()
                NSPasteboard.general.setString(node.content.isEmpty ? node.thinkingTrace : node.content, forType: .string)
            }
            Button(node.role == .user ? "Edit Prompt…" : "Edit / Trim Reply to Save KV RAM…") {
                editMessageDraft = node.content
                clearThinkingOnAssistantEdit = false
                isEditBoxExpandedFull = false
                editingMessageNodeID = node.id
            }
            Button("Branch to New Chat from Here") { store.branchMessageToNewChat(upToNodeID: node.id) }
            Divider()
            Button("Delete Message (Free ~\(node.estimatedTokenFootprint) tok)…", role: .destructive) {
                confirmingDeleteNodeID = node.id
            }
        }
    }

    private func branchPaginationControl(nodeID: UUID, siblingMeta: (currentIndex: Int, totalCount: Int)) -> some View {

            .opacity(siblingMeta.currentIndex == 0 ? 0.35 : 1.0)

    @ViewBuilder
    private func turnMainBody(
        node: ConversationMessageNode,
        isNodeStreaming: Bool,
        opticalFontSize: CGFloat,
        isFullscreenOrWide: Bool
    ) -> some View {
        if !node.attachments.isEmpty {
            ScrollView(.horizontal, showsIndicators: false) {
                HStack(spacing: 8) {
                    ForEach(node.attachments) { att in
                        HStack(spacing: 5) {
                            Image(systemName: att.kind == .downscaledVisionImage ? "photo" : "doc.text").font(.system(size: 10))
                            Text("\(att.title) (\(att.formattedByteSize) · ~\(att.tokenEstimate) tok)")
                                .font(typo.sans(size: 11, weight: .medium).monospacedDigit())
                        }
                        .foregroundColor(tokens.primaryText)
                        .padding(.horizontal, 8).padding(.vertical, 4)
                        .background(tokens.elevatedSurface)
                        .clipShape(RoundedRectangle(cornerRadius: 5, style: .continuous))
                    }
                }
            }
        }

        if !node.thinkingTrace.isEmpty || (!node.isThinkingComplete && isNodeStreaming) {
            thinkingDisclosureView(for: node, isStreaming: isNodeStreaming, opticalFontSize: opticalFontSize)
        }

        if editingMessageNodeID == node.id {
            let approxLines = max(6, editMessageDraft.components(separatedBy: .newlines).reduce(0) { $0 + max(1, ($1.count / 72) + 1) })
            let naturalH = CGFloat(approxLines) * (opticalFontSize * 1.55) + 36.0
            let maxStdH: CGFloat = isFullscreenOrWide ? 640.0 : 460.0
            let editorHeight: CGFloat = isEditBoxExpandedFull
                ? min(920.0, max(maxStdH, naturalH))
                : min(maxStdH, max(200.0, naturalH))

            VStack(alignment: .leading, spacing: 8) {
                HStack {
                    Text(node.role == .user ? "Editing Prompt" : "Editing AI Reply (Trim text to reclaim KV cache memory)")
                        .font(typo.sans(size: 11.5, weight: .semibold))
                        .foregroundColor(tokens.secondaryText)
                    Spacer()
                    Button {
                        withAnimation(.easeOut(duration: 0.14)) { isEditBoxExpandedFull.toggle() }
                    } label: {
                        HStack(spacing: 4) {
                            Image(systemName: isEditBoxExpandedFull ? "arrow.down.right.and.arrow.up.left" : "arrow.up.left.and.arrow.down.right")
                                .font(.system(size: 9.5, weight: .semibold))
                            Text(isEditBoxExpandedFull ? "Standard Height" : "Expand Full Height")
                                .font(typo.sans(size: 11, weight: .medium))
                        }
                        .foregroundColor(tokens.secondaryText)
                        .padding(.horizontal, 8).padding(.vertical, 3)
                        .background(tokens.elevatedSurface)
                        .clipShape(RoundedRectangle(cornerRadius: 5, style: .continuous))
                    }
                    .buttonStyle(.plain)
                }

                MotifStyledTextEditor(text: $editMessageDraft, font: typo.nsFont(size: opticalFontSize), tokens: tokens)
                    .frame(height: editorHeight)
                    .padding(4)
                    .background(tokens.elevatedSurface)
                    .clipShape(RoundedRectangle(cornerRadius: 8, style: .continuous))
                    .overlay(RoundedRectangle(cornerRadius: 8, style: .continuous).strokeBorder(tokens.strongBorder, lineWidth: 1))

                HStack(spacing: 12) {
                    if node.role == .assistant {
                        if !node.thinkingTrace.isEmpty {
                            Toggle("Clear stored thinking box", isOn: $clearThinkingOnAssistantEdit)
                                .font(typo.sans(size: 11.5))
                                .foregroundColor(tokens.secondaryText)
                        }
                        let oldTok = max(1, node.content.count / 4)
                        let newTok = max(1, editMessageDraft.count / 4)
                        let savedTok = max(0, oldTok - newTok)
                        if savedTok > 0 {
                            Text("Saves ~\(savedTok) KV tok")
                                .font(typo.sans(size: 11, weight: .semibold).monospacedDigit())
                                .foregroundColor(tokens.accentAction)
                        }
                    }
                    Spacer()
                    Button("Cancel") { editingMessageNodeID = nil }
                        .font(typo.sans(size: 12))
                        .foregroundColor(tokens.secondaryText)
                        .buttonStyle(.plain)

                    if node.role == .user {
                        Button("Save & Resubmit") {
                            let updated = editMessageDraft
                            editingMessageNodeID = nil
                            store.createEditedMessageBranch(originalNodeID: node.id, updatedText: updated)
                        }
                        .font(typo.sans(size: 12, weight: .semibold))
                        .foregroundColor(tokens.accentForeground)
                        .padding(.horizontal, 10).padding(.vertical, 4)
                        .background(RoundedRectangle(cornerRadius: 5, style: .continuous).fill(tokens.accentAction))
                        .buttonStyle(.plain)
                    } else {
                        Button("Save Trimmed Reply") {
                            let updated = editMessageDraft
                            let clearThink = clearThinkingOnAssistantEdit
                            editingMessageNodeID = nil
                            store.updateAssistantMessageInPlace(nodeID: node.id, updatedText: updated, clearThinking: clearThink)

    private func buildTelemetryFooter(node: ConversationMessageNode, session: ConversationSession, isLatestTurn: Bool) -> (text: String, isWarning: Bool) {
        let answerTok = node.role == .assistant
            ? (node.content.isEmpty ? 0 : max(node.estimatedTokenFootprint, node.metrics?.totalTokens ?? 0))
            : node.estimatedTokenFootprint
        let m = node.metrics
        let tps = m?.tokensPerSecond ?? 0.0
        let cached = m?.promptEvalCachedTokens ?? 0
        let promptTok = m?.promptEvalTokens ?? 0
        let doneReason = m?.doneReason ?? ""

        var parts: [String] = []
        if node.role == .assistant && tps > 0 {
            parts.append(String(format: "%.1f tok/s", tps))
            parts.append("\(answerTok) tok")
        } else {
            parts.append("~\(answerTok) tok")
        }
        if isLatestTurn { parts.append("Chat: \(session.estimatedActiveTokens) tok") }
        if node.role == .assistant && promptTok > 0 && cached > 0 {
            let pct = Int((Double(cached) / Double(promptTok)) * 100.0)
            parts.append("KV Hit \(pct)%")
        }
        if doneReason == "length" { parts.append("Truncated (limit)") }
        return (parts.joined(separator: " · "), doneReason == "length")
    }

    private func sanitizedExtension(for language: String) -> String {
        let lower = language.trimmingCharacters(in: .whitespacesAndNewlines).lowercased()
        switch lower {
        case "swift": return "swift"
        case "python", "py": return "py"
        case "javascript", "js": return "js"
        case "typescript", "ts": return "ts"
        case "rust", "rs": return "rs"
        case "bash", "sh", "zsh", "shell": return "sh"
        case "json": return "json"
        case "yaml", "yml": return "yml"
        case "html": return "html"
        case "css": return "css"
        case "sql": return "sql"
        case "markdown", "md": return "md"
        default: return lower.isEmpty ? "txt" : lower
        }
    }

    private func saveArtifactToDisk(_ artifact: WorkspaceArtifact) {
        let panel = NSSavePanel()
        panel.title = "Save Code Snippet"
        let ext = sanitizedExtension(for: artifact.language)
        panel.nameFieldStringValue = "\(artifact.title.replacingOccurrences(of: " ", with: "_").replacingOccurrences(of: "#", with: "")).\(ext)"
        if panel.runModal() == .OK, let url = panel.url {
            try? artifact.content.write(to: url, atomically: true, encoding: .utf8)
        }
    }

    private func openArtifactInExternalEditor(_ artifact: WorkspaceArtifact) {
        let ext = sanitizedExtension(for: artifact.language)
        let safeTitle = artifact.title.replacingOccurrences(of: " ", with: "_").replacingOccurrences(of: "#", with: "")
        let tempURL = FileManager.default.temporaryDirectory.appendingPathComponent("Motif_\(safeTitle).\(ext)")
        try? artifact.content.write(to: tempURL, atomically: true, encoding: .utf8)
        NSWorkspace.shared.open(tempURL)
    }
}

        let title = personaTitleDraft.trimmingCharacters(in: .whitespacesAndNewlines)

// MARK: - AppKit Window Configurator (Edge-to-Edge Fullscreen, Zero Toolbar Waste & Appearance Sync)
public struct FullSizeWindowConfigurator: NSViewRepresentable {
    public let isDark: Bool

    public init(isDark: Bool) { self.isDark = isDark }

    public func makeNSView(context: Context) -> NSView {
        let view = NSView()
        DispatchQueue.main.async {
            guard let window = view.window else { return }
            window.titleVisibility = .hidden
            window.titlebarAppearsTransparent = true
            window.styleMask.insert(.fullSizeContentView)
            window.toolbar = nil
            window.appearance = NSAppearance(named: isDark ? .darkAqua : .aqua)
        }
        return view
    }

    public func updateNSView(_ nsView: NSView, context: Context) {
        DispatchQueue.main.async {
            guard let window = nsView.window else { return }
            if window.toolbar != nil { window.toolbar = nil }
            if !window.styleMask.contains(.fullSizeContentView) {
                window.styleMask.insert(.fullSizeContentView)
            }
            let targetApp = NSAppearance(named: isDark ? .darkAqua : .aqua)
            if window.appearance != targetApp { window.appearance = targetApp }
        }
    }
}

(
  set -e
  APP_NAME="${APP_NAME:-Motif}"
  DISPLAY_NAME="${DISPLAY_NAME:-Motif}"
  BUNDLE_ID="${BUNDLE_ID:-com.motif.studio.ollama}"
  VERSION="${VERSION:-15.0.0}"
  BUILD_NUMBER="${BUILD_NUMBER:-1500}"

  rm -rf build/dmg_staging/*
  APP_BUNDLE="build/dmg_staging/${APP_NAME}.app"

  mkdir -p "${APP_BUNDLE}/Contents/Resources/Fonts"

  BIN_PATH="$(cd Motif && swift build -c release --arch arm64 --show-bin-path)"
  cp "${BIN_PATH}/Motif" "${APP_BUNDLE}/Contents/MacOS/${APP_NAME}"
  chmod +x "${APP_BUNDLE}/Contents/MacOS/${APP_NAME}"

  if [ -f "build/AppIcon.icns" ]; then
    cp "build/AppIcon.icns" "${APP_BUNDLE}/Contents/Resources/AppIcon.icns"

  if [ -d "Motif/Sources/Motif/Resources/Fonts" ]; then
    cp -R Motif/Sources/Motif/Resources/Fonts/* "${APP_BUNDLE}/Contents/Resources/Fonts/" 2>/dev/null || true
  fi
  if [ -d "${BIN_PATH}/Motif_Motif.bundle" ]; then
    cp -R "${BIN_PATH}/Motif_Motif.bundle" "${APP_BUNDLE}/Contents/Resources/"
  fi

  cat << EOF > "${APP_BUNDLE}/Contents/Info.plist"
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
  <key>CFBundleDevelopmentRegion</key>

  <string>${APP_NAME}</string>
  <key>CFBundleIconFile</key>
  <string>AppIcon</string>
  <key>CFBundleIdentifier</key>
  <string>${BUNDLE_ID}</string>
  <key>CFBundleInfoDictionaryVersion</key>
  <string>6.0</string>
  <key>CFBundleName</key>
  <string>${DISPLAY_NAME}</string>
  <key>CFBundleDisplayName</key>
  <string>${DISPLAY_NAME}</string>
  <key>CFBundlePackageType</key>
  <string>APPL</string>
  <key>CFBundleShortVersionString</key>
  <string>${VERSION}</string>

  <key>NSHighResolutionCapable</key>

  <key>NSSupportsAutomaticGraphicsSwitching</key>
  <true/>
  <key>NSAppTransportSecurity</key>
  <dict>
    <key>NSAllowsLocalNetworking</key>
    <true/>
  </dict>
</dict>
</plist>
EOF

  codesign --force --deep --sign - "${APP_BUNDLE}"
  codesign --verify --verbose "${APP_BUNDLE}"
)

(
  set -e
  APP_NAME="${APP_NAME:-Motif}"
  DISPLAY_NAME="${DISPLAY_NAME:-Motif}"
  VERSION="${VERSION:-15.0.0}"

    -volname "${DISPLAY_NAME}" \
    -srcfolder build/dmg_staging \
    -ov \
    -format UDZO \
    "build/${DMG_NAME}"
  ls -lh "build/${DMG_NAME}"
  printf "\nBuild complete: build/%s\n" "${DMG_NAME}"
)

public enum AppFontFamily: String, CaseIterable, Codable, Identifiable, Sendable {

@MainActor
public enum CustomFontRegistrar {
    public private(set) static var resolvedJakartaName: String = "Plus Jakarta Sans"
    public private(set) static var resolvedInterName: String = "Inter"
    public private(set) static var resolvedOpenSansName: String = "Open Sans"
    public private(set) static var resolvedGeistName: String = "Geist"
    private static var hasRegistered: Bool = false

    public static func registerBundledFonts() {
        guard !hasRegistered else { return }
        hasRegistered = true
        guard let resourceURL = Bundle.main.resourceURL else { return }
        let fontDirs = [
            resourceURL.appendingPathComponent("Fonts"),
            resourceURL.appendingPathComponent("Motif_Motif.bundle/Contents/Resources/Fonts"),
            resourceURL
        ]
        for dir in fontDirs {
            guard let files = try? FileManager.default.contentsOfDirectory(at: dir, includingPropertiesForKeys: nil) else { continue }
            for file in files where file.pathExtension.lowercased() == "ttf" || file.pathExtension.lowercased() == "otf" {
                CTFontManagerRegisterFontsForURL(file as CFURL, .process, nil)
                if let descriptors = CTFontManagerCreateFontDescriptorsFromURL(file as CFURL) as? [CTFontDescriptor],
                   let first = descriptors.first,
                   let familyName = CTFontDescriptorCopyAttribute(first, kCTFontFamilyNameAttribute) as? String {
                    let lower = file.lastPathComponent.lowercased()
                    if lower.contains("jakarta") { resolvedJakartaName = familyName }
                    else if lower.contains("inter") { resolvedInterName = familyName }
                    else if lower.contains("opensans") { resolvedOpenSansName = familyName }
                    else if lower.contains("geist") { resolvedGeistName = familyName }
                }
            }
        }
    }
}

public struct TypographyConfiguration: Codable, Equatable, Sendable {
    public var fontFamily: AppFontFamily = .inter
    public var windowedBaseFontSize: Double = 14.5
    public var fullscreenBaseFontSize: Double = 15.5
    public var lineSpacingMultiplier: Double = 1.64
    public var comfortableReadingWidth: Double = 720.0

    @MainActor
    public var resolvedFamilyName: String {
        switch fontFamily {
        case .plusJakartaSans: return CustomFontRegistrar.resolvedJakartaName
        case .inter: return CustomFontRegistrar.resolvedInterName
        case .openSans: return CustomFontRegistrar.resolvedOpenSansName
        case .geist: return CustomFontRegistrar.resolvedGeistName
        }
    }

    public func effectiveFontSize(forCanvasWidth width: CGFloat, isFullscreen: Bool) -> CGFloat {
        if isFullscreen { return CGFloat(fullscreenBaseFontSize) }
        let progress = min(1.0, max(0.0, (Double(width) - 860.0) / (1440.0 - 860.0)))
        return CGFloat(windowedBaseFontSize + progress * (fullscreenBaseFontSize - windowedBaseFontSize))
    }

    /// Narrow, centered document reading column (680–760pt) so long replies read like editorial paper, never a stretched dashboard.
    public func effectiveReadingWidth(forCanvasWidth width: CGFloat, isFullscreen: Bool) -> CGFloat {
        let baseTarget = min(760.0, max(660.0, CGFloat(comfortableReadingWidth)))
        let target = isFullscreen ? min(760.0, baseTarget + 24.0) : baseTarget
        let available = max(380.0, width - 56.0)
        return min(available, target)
    }

    @MainActor
    public func sans(size: CGFloat, weight: Font.Weight = .regular) -> Font {
        let candidate = resolvedFamilyName
        if NSFont(name: candidate, size: size) != nil {
            return .custom(candidate, size: size).weight(weight)
        }
        return .system(size: size, weight: weight, design: .default)
    }

    @MainActor
    public func nsFont(size: CGFloat) -> NSFont {
        let candidate = resolvedFamilyName
        return NSFont(name: candidate, size: size) ?? NSFont.systemFont(ofSize: size, weight: .regular)
    }

    public func mono(size: CGFloat = 11.5, weight: Font.Weight = .regular) -> Font {
        .system(size: size, weight: weight, design: .monospaced).monospacedDigit()
    }

    @MainActor
    public func wordmark(size: CGFloat = 16.0) -> Font {
        sans(size: size, weight: .semibold)
    }
}

    private func sanitizeIncompleteStreamMarkers(_ input: String, isStreaming: Bool) -> String {
        guard isStreaming else { return input }

        let boldCount = result.components(separatedBy: "**").count - 1

    private func matchHeading(_ trimmed: String) -> (level: Int, text: String)? {
        guard trimmed.hasPrefix("#") else { return nil }
        var count = 0
        for ch in trimmed { if ch == "#" { count += 1 } else { break } }
        guard count >= 1 && count <= 6 else { return nil }
        let remainder = String(trimmed.dropFirst(count))
        guard remainder.isEmpty || remainder.first == " " || remainder.first == "\t" else { return nil }
        var cleaned = remainder.trimmingCharacters(in: .whitespaces)
        while cleaned.hasSuffix("#") { cleaned.removeLast() }
        cleaned = cleaned.trimmingCharacters(in: .whitespaces)
        return cleaned.isEmpty ? nil : (count, cleaned)
    }

        if trimmed.hasSuffix("|") { trimmed.removeLast() }

    private func parseBlocks(from raw: String) -> [MarkdownBlock] {
        let lines = raw.components(separatedBy: .newlines)
        var blocks: [MarkdownBlock] = []
        var paragraphBuffer: [String] = []
        var idx = 0

        func flushParagraph() {
            let joined = paragraphBuffer.joined(separator: "\n").trimmingCharacters(in: .whitespacesAndNewlines)
            if !joined.isEmpty {
                blocks.append(MarkdownBlock(id: "p_\(blocks.count)", kind: .paragraph(joined)))
            }
            paragraphBuffer.removeAll()
        }

        while idx < lines.count {
            let line = lines[idx]
            let trimmed = line.trimmingCharacters(in: .whitespaces)

            if trimmed.hasPrefix("```") || trimmed.hasPrefix("~~~") {
                flushParagraph()
                let fenceChar = trimmed.first!
                let fenceLen = trimmed.prefix(while: { $0 == fenceChar }).count
                let lang = String(trimmed.dropFirst(fenceLen)).trimmingCharacters(in: .whitespaces)
                var codeLines: [String] = []
                idx += 1
                while idx < lines.count {
                    let candidate = lines[idx].trimmingCharacters(in: .whitespaces)
                    if candidate.prefix(while: { $0 == fenceChar }).count >= fenceLen
                        && candidate.drop(while: { $0 == fenceChar }).trimmingCharacters(in: .whitespaces).isEmpty {
                        idx += 1
                        break
                    }
                    codeLines.append(lines[idx])
                    idx += 1
                }
                blocks.append(MarkdownBlock(id: "code_\(blocks.count)", kind: .codeBlock(lang, codeLines.joined(separator: "\n"))))
                continue
            }

            if trimmed.contains("|"), idx + 1 < lines.count, isTableSeparatorLine(lines[idx + 1]) {
                flushParagraph()
                let headers = splitTableRow(trimmed)
                var rows: [[String]] = []
                idx += 2
                while idx < lines.count {
                    let rowLine = lines[idx].trimmingCharacters(in: .whitespaces)
                    if rowLine.isEmpty || !rowLine.contains("|") { break }
                    rows.append(splitTableRow(rowLine))
                    idx += 1
                }
                blocks.append(MarkdownBlock(id: "tbl_\(blocks.count)", kind: .table(headers, rows)))
                continue
            }

            let leadingSpaces = line.prefix(while: { $0 == " " || $0 == "\t" }).reduce(0) { $0 + ($1 == "\t" ? 4 : 1) }
            let indentLevel = leadingSpaces / 2

            if trimmed.isEmpty {
                flushParagraph()
            } else if trimmed == "---" || trimmed == "***" || trimmed == "___" {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "hr_\(blocks.count)", kind: .horizontalRule))
            } else if let heading = matchHeading(trimmed) {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "h_\(blocks.count)", kind: .heading(heading.level, heading.text)))
            } else if trimmed.hasPrefix("- ") || trimmed.hasPrefix("* ") || trimmed.hasPrefix("• ") || trimmed.hasPrefix("+ ") {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "b_\(blocks.count)", kind: .bulletItem(indentLevel, String(trimmed.dropFirst(2)))))
            } else if trimmed.hasPrefix("> ") || trimmed == ">" {
                flushParagraph()
                let quoteBody = trimmed.count > 2 ? String(trimmed.dropFirst(2)) : ""
                if !quoteBody.isEmpty {
                    blocks.append(MarkdownBlock(id: "q_\(blocks.count)", kind: .blockquote(quoteBody)))
                }
            } else if let num = matchNumberedList(trimmed) {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "n_\(blocks.count)", kind: .numberedItem(indentLevel, num.number, num.rest)))
            } else {
                paragraphBuffer.append(line)
            }
            idx += 1
        }
        flushParagraph()
        return blocks
    }

    private func matchNumberedList(_ line: String) -> (number: String, rest: String)? {
        guard let sepIdx = line.firstIndex(of: ".") ?? line.firstIndex(of: ")") else { return nil }
        let prefix = String(line[..<sepIdx])
        guard !prefix.isEmpty, prefix.allSatisfy({ $0.isNumber }) else { return nil }
        let afterSep = line.index(after: sepIdx)
        guard afterSep < line.endIndex, line[afterSep] == " " else { return nil }
        return ("\(prefix).", String(line[line.index(after: afterSep)...]))
    }

    private func handleDroppedProviders(_ providers: [NSItemProvider]) -> Bool {
        guard store.activeSession != nil else { return false }
        var handled = false
        for provider in providers {
            if provider.hasItemConformingToTypeIdentifier(UTType.fileURL.identifier) {
                handled = true
                provider.loadItem(forTypeIdentifier: UTType.fileURL.identifier, options: nil) { item, _ in
                    var resolvedURL: URL? = nil
                    if let data = item as? Data, let str = String(data: data, encoding: .utf8) {
                        resolvedURL = URL(string: str)
                    } else if let url = item as? URL {
                        resolvedURL = url
                    }
                    if let url = resolvedURL {
                        Task { @MainActor in store.attachFilesFromURLs([url]) }
                    }
                }
            } else if provider.hasItemConformingToTypeIdentifier(UTType.image.identifier) {
                handled = true
                provider.loadDataRepresentation(forTypeIdentifier: UTType.image.identifier) { data, _ in
                    if let data = data {
                        Task { @MainActor in store.handleClipboardImagePaste(data) }
                    }
                }
            }
        }
        return handled
    }

    // MARK: - Quiet Minimalist Empty New-Chat Canvas (Zero Control-Panel Clutter)
    @ViewBuilder
    private func quietNewChatCanvas(session: ConversationSession, columnWidth: CGFloat) -> some View {
        let activeSessionIdx = store.sessions.firstIndex(where: { $0.id == session.id })
        let model = store.activeModelDescriptor
        let isLoading = store.isLoadingModelWeights
        let isHoldingDone = store.isHoldingLoadCompletionState
        let snap = store.memorySnapshot
        let loadFraction = isHoldingDone ? 1.0 : min(0.99, max(0.02, snap.loadProgressAgainstTarget))
        let pct = isHoldingDone ? 100 : Int(loadFraction * 100.0)

        VStack(spacing: 20) {
            Spacer()

            VStack(spacing: 10) {
                Text(session.title == "New Chat" ? "What can I help with?" : session.title)
                    .font(typo.sans(size: 24, weight: .medium))
                    .tracking(-0.45)
                    .foregroundColor(tokens.primaryText)
                    .multilineTextAlignment(.center)

                if let m = model {
                    HStack(spacing: 8) {
                        Circle()
                            .fill(m.residency == .activeInUnifiedRAM || isHoldingDone ? tokens.statusOnline : tokens.mutedText.opacity(0.6))
                            .frame(width: 6, height: 6)

                        Text("\(m.shortDisplayName) · \(session.parameters.numCtx / 1024)k context · \(m.residency == .activeInUnifiedRAM ? "In RAM" : m.formattedSize)")
                            .font(typo.sans(size: 12.5).monospacedDigit())
                            .foregroundColor(tokens.secondaryText)

                        if m.residency == .dormantOnDisk && !isLoading {
                            Button("Preload") { store.selectAndLoadModel(m) }
                                .font(typo.sans(size: 12, weight: .medium))
                                .foregroundColor(tokens.primaryText)
                                .padding(.horizontal, 8).padding(.vertical, 2.5)
                                .background(tokens.codeSurface)
                                .clipShape(RoundedRectangle(cornerRadius: 6, style: .continuous))
                                .overlay(RoundedRectangle(cornerRadius: 6, style: .continuous).strokeBorder(tokens.hairline, lineWidth: 1))
                                .buttonStyle(.plain)
                        } else if m.residency == .activeInUnifiedRAM && !isLoading {
                            Button("Unload") { Task { await store.unloadModelFromMemory(m) } }
                                .font(typo.sans(size: 12, weight: .regular))
                                .foregroundColor(tokens.mutedText)
                                .buttonStyle(.plain)
                        }
                    }
                } else {
                    Button("Open Settings to pull a local model…") {
                        store.settingsRequestedTab = "Models & Storage"
                        store.isSettingsModalPresented = true
                    }
                    .font(typo.sans(size: 13, weight: .medium))
                    .foregroundColor(tokens.secondaryText)
                    .buttonStyle(.plain)
                }
            }

            if isLoading {
                VStack(spacing: 6) {
                    HStack {
                        Text(isHoldingDone ? "\(store.loadingModelName) ready in RAM" : "Loading \(store.loadingModelName)…")
                            .font(typo.sans(size: 12, weight: .medium))
                            .foregroundColor(isHoldingDone ? tokens.statusOnline : tokens.secondaryText)
                        Spacer()
                        Text("\(pct)%")
                            .font(typo.sans(size: 12, weight: .medium).monospacedDigit())
                            .foregroundColor(isHoldingDone ? tokens.statusOnline : tokens.primaryText)
                    }
                    ProgressView(value: loadFraction)
                        .tint(isHoldingDone ? tokens.statusOnline : tokens.accentAction)
                }
                .frame(maxWidth: 320)
                .padding(.top, 4)
            }

            // Quiet preset chips (replaces the heavy pre-flight cockpit while keeping 1-click access)
            if let sIdx = activeSessionIdx, !store.presets.isEmpty {
                let sessionBinding = $store.sessions[sIdx]
                HStack(spacing: 8) {
                    ForEach(store.presets.prefix(4)) { preset in
                        let isMatch = sessionBinding.wrappedValue.parameters == preset.parameters
                        Button {
                            sessionBinding.wrappedValue.parameters = preset.parameters
                            sessionBinding.wrappedValue.systemPrompt = preset.systemPrompt
                            store.recordActiveModelParameterPreference()
                            if store.supervisor.kvCacheTypeEnv != preset.parameters.kvCacheQuant {
                                store.applyKVCacheQuantizationLive(preset.parameters.kvCacheQuant)
                            } else {
                                store.recalculateHardwareTelemetry(sampleKernelRSS: false)
                            }
                        } label: {
                            Text(preset.name)
                                .font(typo.sans(size: 12, weight: isMatch ? .medium : .regular))
                                .foregroundColor(isMatch ? tokens.primaryText : tokens.secondaryText)
                                .padding(.horizontal, 11).padding(.vertical, 5.5)
                                .background(
                                    RoundedRectangle(cornerRadius: 8, style: .continuous)
                                        .fill(isMatch ? tokens.codeSurface : Color.clear)
                                )
                                .overlay(
                                    RoundedRectangle(cornerRadius: 8, style: .continuous)
                                        .strokeBorder(isMatch ? tokens.strongBorder : tokens.hairline, lineWidth: 1)
                                )
                        }
                        .buttonStyle(.plain)
                        .help(preset.summary)
                    }
                }
                .padding(.top, 4)
            }

            Spacer()
        }
        .frame(maxWidth: columnWidth)
        .padding(.bottom, max(120, composerMeasuredHeight + 16))
        .frame(maxWidth: .infinity, maxHeight: .infinity)
    }

    // MARK: - Centered Document-Width Conversation Stream
    private func conversationStreamCanvas(
        session: ConversationSession,
        columnWidth: CGFloat,
        horizontalPadding: CGFloat,
        opticalFontSize: CGFloat,
        isFullscreenOrWide: Bool
    ) -> some View {
        let thread = session.activeBranchThread
        let lastTurnID = thread.last?.id
        let dynamicBottomClearance = max(144.0, composerMeasuredHeight + 24.0)
        let jumpButtonBottomOffset = max(112.0, composerMeasuredHeight + 6.0)

        return ScrollViewReader { proxy in
            ScrollView(.vertical, showsIndicators: true) {
                LazyVStack(alignment: .leading, spacing: 26) {
                    ForEach(thread) { node in
                        ConversationTurnRow(
                            store: store,
                            node: node,
                            session: session,
                            isLatestTurn: node.id == lastTurnID,
                            opticalFontSize: opticalFontSize,
                            isFullscreenOrWide: isFullscreenOrWide,
                            columnWidth: columnWidth,
                            tokens: tokens,
                            editingMessageNodeID: $editingMessageNodeID,
                            editMessageDraft: $editMessageDraft,
                            clearThinkingOnAssistantEdit: $clearThinkingOnAssistantEdit,
                            isEditBoxExpandedFull: $isEditBoxExpandedFull,
                            confirmingDeleteNodeID: $confirmingDeleteNodeID
                        )
                        .id(node.id)
                    }

                    Color.clear
                        .frame(height: dynamicBottomClearance)
                        .id("STREAM_BOTTOM_ANCHOR")
                }
                .background(
                    ScrollViewBottomObserver(
                        isScrolledAwayFromBottom: $store.isScrolledAwayFromBottom,
                        isDark: tokens.isEffectivelyDark
                    ).frame(width: 1, height: 1)
                )
                .frame(maxWidth: columnWidth, alignment: .leading)
                .padding(.horizontal, horizontalPadding)
                .padding(.top, 26)
                .frame(maxWidth: .infinity)
            }
            .onChange(of: store.streamScrollPulse) { _, _ in
                if store.isStreamingGeneration && !store.isScrolledAwayFromBottom && !store.isAutoScrollPaused {
                    proxy.scrollTo("STREAM_BOTTOM_ANCHOR", anchor: .bottom)
                }
            }
            .onChange(of: store.pendingScrollToMessageID) { _, targetID in
                if let id = targetID {
                    withAnimation(.spring(response: 0.28, dampingFraction: 0.86)) { proxy.scrollTo(id, anchor: .center) }
                    store.pendingScrollToMessageID = nil
                }
            }
            .overlay(alignment: .bottom) {
                if store.isScrolledAwayFromBottom && !store.isComposerCanvasExpanded && store.matchedSlashPersonas.isEmpty {
                    HStack {
                        Spacer()
                        HighContrastJumpToLatestButton(tokens: tokens) {
                            store.isScrolledAwayFromBottom = false
                            store.isAutoScrollPaused = false
                            withAnimation(.easeOut(duration: 0.18)) { proxy.scrollTo("STREAM_BOTTOM_ANCHOR", anchor: .bottom) }
                        }
                    }
                    .frame(maxWidth: columnWidth)
                    .padding(.horizontal, horizontalPadding)
                    .padding(.bottom, jumpButtonBottomOffset)
                    .transition(.scale(scale: 0.92).combined(with: .opacity))
                }
            }
        }
    }
}

    private var assistantTurnBody: some View {
        VStack(alignment: .leading, spacing: 10) {
            if !node.thinkingTrace.isEmpty || (!node.isThinkingComplete && isNodeStreaming) {
                borderlessThinkingDisclosure
            }

                if store.isLoadingModelWeights {
                    let isHoldingDone = store.isHoldingLoadCompletionState

                             ? "\(store.loadingModelName) loaded into Unified Memory (100%)"

                            .frame(maxWidth: 240)

                    Text("Reading conversation context…")
                        .font(typo.sans(size: 13))

    private func buildTelemetryFooter() -> (text: String, isWarning: Bool) {
        let answerTok = node.content.isEmpty ? 0 : max(node.estimatedTokenFootprint, node.metrics?.totalTokens ?? 0)
        let m = node.metrics
        let tps = m?.tokensPerSecond ?? 0.0
        let cached = m?.promptEvalCachedTokens ?? 0
        let promptTok = m?.promptEvalTokens ?? 0
        let doneReason = m?.doneReason ?? ""

        var parts: [String] = []
        if tps > 0 {
            parts.append(String(format: "%.1f tok/s", tps))
            parts.append("\(answerTok) tok")
        } else {
            parts.append("~\(answerTok) tok")
        }
        if promptTok > 0 && cached > 0 {
            let pct = Int((Double(cached) / Double(promptTok)) * 100.0)
            parts.append("KV \(pct)%")
        }
        if doneReason == "length" { parts.append("Truncated") }
        return (parts.joined(separator: " · "), doneReason == "length")
    }

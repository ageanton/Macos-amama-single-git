    private static func linearize(_ channel: Double) -> Double {

@MainActor
public enum CustomFontRegistrar {
    public private(set) static var resolvedJakartaName: String = "Plus Jakarta Sans"
    public private(set) static var resolvedInterName: String = "Inter"
    public private(set) static var resolvedOpenSansName: String = "Open Sans"
    public private(set) static var resolvedGeistName: String = "Geist"
    private static var hasRegistered: Bool = false

    public static func registerBundledFonts() {
        guard !hasRegistered else { return }
        hasRegistered = true
        guard let resourceURL = Bundle.main.resourceURL else { return }
        let fontDirs = [
            resourceURL.appendingPathComponent("Fonts"),
            resourceURL.appendingPathComponent("Motif_Motif.bundle/Contents/Resources/Fonts"),
            resourceURL
        ]
        for dir in fontDirs {
            guard let files = try? FileManager.default.contentsOfDirectory(at: dir, includingPropertiesForKeys: nil) else { continue }
            for file in files where file.pathExtension.lowercased() == "ttf" || file.pathExtension.lowercased() == "otf" {
                CTFontManagerRegisterFontsForURL(file as CFURL, .process, nil)
                if let descriptors = CTFontManagerCreateFontDescriptorsFromURL(file as CFURL) as? [CTFontDescriptor],
                   let first = descriptors.first,
                   let familyName = CTFontDescriptorCopyAttribute(first, kCTFontFamilyNameAttribute) as? String {
                    let lower = file.lastPathComponent.lowercased()
                    if lower.contains("jakarta") { resolvedJakartaName = familyName }
                    else if lower.contains("inter") { resolvedInterName = familyName }
                    else if lower.contains("opensans") { resolvedOpenSansName = familyName }
                    else if lower.contains("geist") { resolvedGeistName = familyName }
                }
            }
        }
    }
}

public struct TypographyConfiguration: Codable, Equatable, Sendable {
    public var fontFamily: AppFontFamily = .inter
    public var windowedBaseFontSize: Double = 14.5
    public var fullscreenBaseFontSize: Double = 15.5
    public var lineSpacingMultiplier: Double = 1.62
    public var comfortableReadingWidth: Double = 820.0

    @MainActor
    public var resolvedFamilyName: String {
        switch fontFamily {
        case .plusJakartaSans: return CustomFontRegistrar.resolvedJakartaName
        case .inter: return CustomFontRegistrar.resolvedInterName
        case .openSans: return CustomFontRegistrar.resolvedOpenSansName
        case .geist: return CustomFontRegistrar.resolvedGeistName
        }
    }

    public func effectiveFontSize(forCanvasWidth width: CGFloat, isFullscreen: Bool) -> CGFloat {
        if isFullscreen { return CGFloat(fullscreenBaseFontSize) }
        let progress = min(1.0, max(0.0, (Double(width) - 860.0) / (1480.0 - 860.0)))
        return CGFloat(windowedBaseFontSize + progress * (fullscreenBaseFontSize - windowedBaseFontSize))
    }

    /// Responsive desktop reading column: generous ~820pt windowed, expanding up to ~1040pt on wide/fullscreen Macs.
    public func effectiveReadingWidth(forCanvasWidth width: CGFloat, isFullscreen: Bool) -> CGFloat {
        let horizontalMargin: CGFloat = width < 960 ? 44.0 : 88.0
        let available = max(460.0, width - horizontalMargin)
        let target: CGFloat
        if isFullscreen || width >= 1360 {
            target = min(1040.0, max(CGFloat(comfortableReadingWidth) + 140.0, width * 0.74))
        } else {
            target = min(880.0, max(CGFloat(comfortableReadingWidth), width * 0.80))
        }
        return min(available, target)
    }

    @MainActor
    public func sans(size: CGFloat, weight: Font.Weight = .regular) -> Font {
        let candidate = resolvedFamilyName
        if NSFont(name: candidate, size: size) != nil {
            return .custom(candidate, size: size).weight(weight)
        }
        return .system(size: size, weight: weight, design: .default)
    }

    @MainActor
    public func nsFont(size: CGFloat, weight: NSFont.Weight = .regular) -> NSFont {
        let candidate = resolvedFamilyName
        if let base = NSFont(name: candidate, size: size) {
            if weight == .regular { return base }
            let desc = base.fontDescriptor.addingAttributes([
                .traits: [NSFontDescriptor.TraitKey.weight: weight]
            ])
            return NSFont(descriptor: desc, size: size) ?? base
        }

// MARK: - Native 120Hz AppKit Scroll Controller (Isolated From Parent Re-Renders)
@MainActor
public final class ChatScrollController: ObservableObject {
    @Published public var showJumpToTop: Bool = false
    @Published public var showJumpToBottom: Bool = false
    public weak var scrollView: NSScrollView?

    public init() {}

    public func reset() {
        showJumpToTop = false
        showJumpToBottom = false
        scrollView = nil
    }

    public func scrollToTop(animated: Bool = true) {
        guard let sv = scrollView, let docView = sv.documentView else { return }
        let clipH = sv.contentView.bounds.height
        let docH = docView.frame.height
        let targetY: CGFloat = docView.isFlipped ? 0 : max(0, docH - clipH)
        performSmoothScroll(in: sv, toY: targetY, animated: animated, settleToBottom: false)
    }

    public func scrollToBottom(animated: Bool = true) {
        guard let sv = scrollView, let docView = sv.documentView else { return }
        let clipH = sv.contentView.bounds.height
        let docH = docView.frame.height
        let targetY: CGFloat = docView.isFlipped ? max(0, docH - clipH) : 0
        performSmoothScroll(in: sv, toY: targetY, animated: animated, settleToBottom: true)
    }

    private func performSmoothScroll(in sv: NSScrollView, toY: CGFloat, animated: Bool, settleToBottom: Bool) {
        let clipView = sv.contentView
        let targetOrigin = NSPoint(x: clipView.bounds.origin.x, y: toY)
        if animated {
            NSAnimationContext.runAnimationGroup { ctx in
                ctx.duration = 0.22
                ctx.timingFunction = CAMediaTimingFunction(name: .easeOut)
                clipView.animator().setBoundsOrigin(targetOrigin)
            } completionHandler: {
                Task { @MainActor [weak self] in
                    sv.reflectScrolledClipView(clipView)
                    if settleToBottom, let self = self {
                        // Settle exact bottom in case LazyVStack adjusted offscreen estimates during flight
                        self.scrollToBottom(animated: false)
                    }
                }
            }
        } else {
            clipView.setBoundsOrigin(targetOrigin)
            sv.reflectScrolledClipView(clipView)
        }
    }
}

public struct ChatScrollBridgeView: NSViewRepresentable {
    public let controller: ChatScrollController
    public let isDark: Bool
    public let onBottomAwayChanged: (Bool) -> Void

    public init(
        controller: ChatScrollController,
        isDark: Bool,

    ) {

        self.onBottomAwayChanged = onBottomAwayChanged
    }

    public func makeNSView(context: Context) -> BridgeNSView {

        view.controller = controller
        view.isDark = isDark
        view.onBottomAwayChanged = onBottomAwayChanged
        return view
    }

    public func updateNSView(_ nsView: BridgeNSView, context: Context) {
        nsView.controller = controller
        nsView.isDark = isDark
        nsView.onBottomAwayChanged = onBottomAwayChanged
        nsView.syncScrollerAppearance()
        nsView.evaluateScrollPosition()
    }

    public final class BridgeNSView: NSView {
        public weak var controller: ChatScrollController?
        public var isDark: Bool = false
        public var onBottomAwayChanged: ((Bool) -> Void)?
        private var observers: [NSObjectProtocol] = []
        private weak var trackedScrollView: NSScrollView?
        private var lastTopVisible: Bool = false
        private var lastBottomVisible: Bool = false

        public override func viewDidMoveToWindow() {
            super.viewDidMoveToWindow()
            DispatchQueue.main.async { [weak self] in self?.attachToEnclosingScrollView() }
        }

        public func syncScrollerAppearance() {
            guard let sv = trackedScrollView else { return }
            if sv.scrollerStyle != .overlay { sv.scrollerStyle = .overlay }
            sv.autohidesScrollers = true
            sv.drawsBackground = false
            let targetApp = NSAppearance(named: isDark ? .darkAqua : .aqua)
            if sv.appearance != targetApp { sv.appearance = targetApp }
        }

        private func attachToEnclosingScrollView() {
            observers.forEach { NotificationCenter.default.removeObserver($0) }
            observers.removeAll()
            var candidate: NSView? = self
            var foundScroll: NSScrollView? = nil
            while let cur = candidate {
                if let sv = cur as? NSScrollView { foundScroll = sv; break }
                candidate = cur.superview
            }
            guard let scrollView = foundScroll else { return }
            trackedScrollView = scrollView
            Task { @MainActor [weak self] in
                self?.controller?.scrollView = scrollView
            }
            syncScrollerAppearance()
            scrollView.contentView.postsBoundsChangedNotifications = true
            scrollView.contentView.postsFrameChangedNotifications = true
            scrollView.documentView?.postsFrameChangedNotifications = true

            let nc = NotificationCenter.default
            observers = [
                nc.addObserver(forName: NSView.boundsDidChangeNotification, object: scrollView.contentView, queue: .main) { [weak self] _ in self?.evaluateScrollPosition() },
                nc.addObserver(forName: NSView.frameDidChangeNotification, object: scrollView.contentView, queue: .main) { [weak self] _ in self?.evaluateScrollPosition() },
                nc.addObserver(forName: NSView.frameDidChangeNotification, object: scrollView.documentView, queue: .main) { [weak self] _ in self?.evaluateScrollPosition() }
            ]
            evaluateScrollPosition()
        }

        public func evaluateScrollPosition() {
            guard let sv = trackedScrollView, let docView = sv.documentView else { return }
            let clipBounds = sv.contentView.bounds
            let viewportH = max(240.0, clipBounds.height)
            let docH = docView.frame.height
            let maxScrollable = max(0.0, docH - viewportH)

            if maxScrollable < 240.0 {
                publishFlags(top: false, bottom: false)
                return
            }

            let distanceFromTop: CGFloat
            let distanceFromBottom: CGFloat
            if docView.isFlipped {
                distanceFromTop = max(0.0, clipBounds.minY)
                distanceFromBottom = max(0.0, docH - clipBounds.maxY)
            } else {
                distanceFromBottom = max(0.0, clipBounds.minY)
                distanceFromTop = max(0.0, docH - clipBounds.maxY)
            }

            let showTop = lastTopVisible ? (distanceFromTop > 140.0) : (distanceFromTop > 320.0)
            let showBottom = lastBottomVisible ? (distanceFromBottom > 120.0) : (distanceFromBottom > 280.0)
            publishFlags(top: showTop, bottom: showBottom)
        }

        private func publishFlags(top: Bool, bottom: Bool) {
            guard top != lastTopVisible || bottom != lastBottomVisible else { return }
            lastTopVisible = top
            lastBottomVisible = bottom
            onBottomAwayChanged?(bottom)
            Task { @MainActor [weak self] in
                guard let ctrl = self?.controller else { return }
                if ctrl.showJumpToTop != top { ctrl.showJumpToTop = top }
                if ctrl.showJumpToBottom != bottom { ctrl.showJumpToBottom = bottom }
            }
        }

        deinit { observers.forEach { NotificationCenter.default.removeObserver($0) } }
    }
}

// MARK: - Currency-Safe LaTeX Normalizer (Never Mangles "$5 to $10"!)
public enum LaTeXAndSymbolNormalizer {
    private static let mathHintChars = CharacterSet(charactersIn: "\\^_=+*/<>±×÷≈≠≤≥∑∫√αβγδεθλμπσωΔΣΩ")

    public static func normalizeProse(_ input: String) -> String {
        guard input.contains("$") || input.contains("\\") else { return input }
        var s = input

        s = s.replacingOccurrences(of: "$$", with: "")
        s = s.replacingOccurrences(of: "\\[", with: "")
        s = s.replacingOccurrences(of: "\\]", with: "")
        s = s.replacingOccurrences(of: "\\(", with: "")
        s = s.replacingOccurrences(of: "\\)", with: "")

        let wrappers = [
            "\\\\textbf\\{([^\\}]*)\\}": "**$1**",
            "\\\\textit\\{([^\\}]*)\\}": "*$1*",
            "\\\\text\\{([^\\}]*)\\}": "$1",
            "\\\\mathrm\\{([^\\}]*)\\}": "$1",
            "\\\\mathbf\\{([^\\}]*)\\}": "**$1**",
            "\\\\sqrt\\{([^\\}]*)\\}": "√($1)",
            "\\\\frac\\{([^\\}]*)\\}\\{([^\\}]*)\\}": "($1/$2)"
        ]
        for (pattern, template) in wrappers {
            if let regex = try? NSRegularExpression(pattern: pattern) {
                s = regex.stringByReplacingMatches(in: s, range: NSRange(s.startIndex..., in: s), withTemplate: template)
            }
        }

        if s.contains("$"), let inlineMathRegex = try? NSRegularExpression(pattern: "\\$(?!\\s)([^$\\n]+?)(?<!\\s)\\$") {
            let ns = s as NSString
            let matches = inlineMathRegex.matches(in: s, range: NSRange(location: 0, length: ns.length))
            if !matches.isEmpty {
                var rebuilt = s
                for match in matches.reversed() where match.numberOfRanges >= 2 {
                    guard let fullRange = Range(match.range(at: 0), in: rebuilt),
                          let innerRange = Range(match.range(at: 1), in: rebuilt) else { continue }
                    let rawInner = String(rebuilt[innerRange])

                    let startsWithNumberAndSpace = rawInner.first?.isNumber == true && rawInner.contains(" ")
                    let hasMathHint = rawInner.unicodeScalars.contains(where: { mathHintChars.contains($0) })
                    let isSingleVariable = rawInner.count <= 2 && rawInner.allSatisfy({ $0.isLetter })
                    if startsWithNumberAndSpace || (!hasMathHint && !isSingleVariable) {
                        continue
                    }

                    var mathInner = rawInner
                    let mathSubSup: [(String, String)] = [
                        ("^2", "²"), ("^3", "³"), ("^n", "ⁿ"),
                        ("_0", "₀"), ("_1", "₁"), ("_2", "₂"), ("_i", "ᵢ"), ("_n", "ₙ")
                    ]
                    for (k, v) in mathSubSup { mathInner = mathInner.replacingOccurrences(of: k, with: v) }
                    rebuilt.replaceSubrange(fullRange, with: mathInner)
                }
                s = rebuilt
            }
        }

        let replacements: [(String, String)] = [
            ("\\times", "×"), ("\\div", "÷"), ("\\pm", "±"), ("\\mp", "∓"),
            ("\\cdot", "·"), ("\\approx", "≈"), ("\\neq", "≠"), ("\\ne", "≠"),
            ("\\leq", "≤"), ("\\le", "≤"), ("\\geq", "≥"), ("\\ge", "≥"),
            ("\\rightarrow", "→"), ("\\to", "→"), ("\\leftarrow", "←"),
            ("\\Rightarrow", "⇒"), ("\\Leftarrow", "⇐"), ("\\leftrightarrow", "↔"),
            ("\\infty", "∞"), ("\\partial", "∂"), ("\\nabla", "∇"),
            ("\\forall", "∀"), ("\\exists", "∃"), ("\\in", "∈"), ("\\notin", "∉"),
            ("\\subset", "⊂"), ("\\subseteq", "⊆"), ("\\cup", "∪"), ("\\cap", "∩"),
            ("\\sum", "∑"), ("\\prod", "∏"), ("\\int", "∫"),
            ("\\alpha", "α"), ("\\beta", "β"), ("\\gamma", "γ"), ("\\delta", "δ"),
            ("\\epsilon", "ε"), ("\\zeta", "ζ"), ("\\eta", "η"), ("\\theta", "θ"),
            ("\\lambda", "λ"), ("\\mu", "μ"), ("\\pi", "π"), ("\\rho", "ρ"),
            ("\\sigma", "σ"), ("\\tau", "τ"), ("\\phi", "φ"), ("\\chi", "χ"),
            ("\\psi", "ψ"), ("\\omega", "ω"), ("\\Delta", "Δ"), ("\\Sigma", "Σ"),
            ("\\Omega", "Ω"), ("\\Gamma", "Γ"), ("\\Lambda", "Λ"), ("\\Theta", "Θ")
        ]
        for (latex, unicode) in replacements { s = s.replacingOccurrences(of: latex, with: unicode) }
        return s
    }
}

// MARK: - Markdown Block AST
public enum MarkdownBlockKind: Equatable, Sendable {
    case paragraph(String)
    case heading(Int, String)
    case bulletItem(Int, String)
    case numberedItem(Int, String, String)
    case blockquote(String)
    case horizontalRule
    case codeBlock(String, String)
    case table([String], [[String]])
}

public struct MarkdownBlock: Identifiable, Equatable, Sendable {
    public let id: String
    public let kind: MarkdownBlockKind
}

@MainActor
public final class MarkdownParseCache {
    public static let shared = MarkdownParseCache()
    private var exactCache: [String: [MarkdownBlock]] = [:]

    public func blocks(for raw: String, isStreaming: Bool, builder: () -> [MarkdownBlock]) -> [MarkdownBlock] {
        if !isStreaming, let cached = exactCache[raw] {
            return cached
        }
        let computed = builder()
        if !isStreaming {
            if exactCache.count > 250 { exactCache.removeAll(keepingCapacity: true) }
            exactCache[raw] = computed
        }
        return computed
    }
}

// MARK: - Lightweight Multi-Language Syntax Highlighter
public enum CodeSyntaxHighlighter {
    private static let keywords: Set<String> = [
        "func", "let", "var", "struct", "class", "enum", "actor", "protocol", "extension",
        "import", "return", "if", "else", "switch", "case", "default", "for", "while",
        "guard", "try", "catch", "throw", "throws", "async", "await", "public", "private",
        "internal", "static", "self", "nil", "true", "false", "init", "deinit", "some", "any",
        "def", "from", "as", "with", "yield", "lambda", "None", "True", "False", "pass", "elif",
        "const", "function", "export", "interface", "type", "new", "null", "undefined",
        "fn", "mut", "pub", "impl", "match", "use", "mod", "crate", "SELECT", "FROM", "WHERE",
        "INSERT", "INTO", "VALUES", "CREATE", "TABLE", "DELETE", "UPDATE", "SET", "ORDER", "BY"
    ]

    public static func highlight(code: String, isDark: Bool, baseColor: Color) -> AttributedString {
        var result = AttributedString(code)
        result.foregroundColor = baseColor
        result.font = .system(size: 12.5, weight: .regular, design: .monospaced)

        let kwColor = isDark ? Color(red: 0.78, green: 0.56, blue: 0.98) : Color(red: 0.58, green: 0.14, blue: 0.58)
        let strColor = isDark ? Color(red: 0.96, green: 0.54, blue: 0.48) : Color(red: 0.76, green: 0.16, blue: 0.12)
        let commentColor = isDark ? Color(red: 0.48, green: 0.54, blue: 0.62) : Color(red: 0.42, green: 0.47, blue: 0.52)
        let numColor = isDark ? Color(red: 0.48, green: 0.78, blue: 0.96) : Color(red: 0.11, green: 0.36, blue: 0.74)

        let nsCode = code as NSString
        let fullNSRange = NSRange(location: 0, length: nsCode.length)

        func applyPattern(_ pattern: String, color: Color, options: NSRegularExpression.Options = [], filter: ((String) -> Bool)? = nil) {
            guard let regex = try? NSRegularExpression(pattern: pattern, options: options) else { return }
            for match in regex.matches(in: code, range: fullNSRange) {
                guard let swiftRange = Range(match.range, in: code) else { continue }
                if let filter = filter, !filter(String(code[swiftRange])) { continue }
                if let attrRange = Range(swiftRange, in: result) {
                    result[attrRange].foregroundColor = color
                }
            }
        }

        applyPattern("\\b[A-Za-z_][A-Za-z0-9_]*\\b", color: kwColor) { keywords.contains($0) }
        applyPattern("\\b\\d+(\\.\\d+)?\\b", color: numColor)
        applyPattern("\"([^\"\\\\]|\\\\.)*\"|'([^'\\\\]|\\\\.)*'", color: strColor)
        applyPattern("(//[^\\n]*|#[^\\n]*)", color: commentColor)

        return result
    }
}

private enum RenderedMarkdownSection: Identifiable {
    case proseGroup(String, [MarkdownBlock])
    case horizontalRule(String)
    case codeBlock(String, String, String)
    case table(String, [String], [[String]])

    var id: String {
        switch self {
        case .proseGroup(let id, _): return id
        case .horizontalRule(let id): return id
        case .codeBlock(let id, _, _): return id
        case .table(let id, _, _): return id
        }
    }
}

// MARK: - Unified Pure-SwiftUI Markdown View (Continuous Cross-Paragraph Selection + Zero I-Beam Glitches + Equatable)
public struct StreamSafeMarkdownView: View, Equatable {
    public let rawMarkdown: String
    public let isStreaming: Bool
    public let effectiveFontSize: CGFloat
    public let tokens: MotifTokens
    public let typo: TypographyConfiguration
    public var useMonospaceBody: Bool = false
    public var overrideTextColor: Color? = nil
    public let onOpenCodeInInspector: (String, String) -> Void

    public static func == (lhs: StreamSafeMarkdownView, rhs: StreamSafeMarkdownView) -> Bool {
        lhs.rawMarkdown == rhs.rawMarkdown
            && lhs.isStreaming == rhs.isStreaming
            && lhs.effectiveFontSize == rhs.effectiveFontSize
            && lhs.tokens == rhs.tokens
            && lhs.typo == rhs.typo
            && lhs.useMonospaceBody == rhs.useMonospaceBody
            && lhs.overrideTextColor == rhs.overrideTextColor
    }

    public var body: some View {
        let blocks = MarkdownParseCache.shared.blocks(for: rawMarkdown, isStreaming: isStreaming) {
            parseBlocks(from: sanitizeIncompleteStreamMarkers(rawMarkdown, isStreaming: isStreaming))
        }
        let sections = groupContiguousProse(blocks)
        let lineGap = CGFloat(effectiveFontSize * (useMonospaceBody ? 0.34 : (typo.lineSpacingMultiplier - 1.0)))

        VStack(alignment: .leading, spacing: useMonospaceBody ? 10 : 14) {
            ForEach(sections) { section in
                switch section {
                case .proseGroup(_, let proseBlocks):
                    Text(buildUnifiedAttributedString(for: proseBlocks))
                        .lineSpacing(lineGap)
                        .textSelection(.enabled)
                        .fixedSize(horizontal: false, vertical: true)
                        .frame(maxWidth: .infinity, alignment: .leading)

                case .horizontalRule:
                    HairlineDivider(tokens: tokens)
                        .padding(.vertical, 4)

                case .codeBlock(_, let lang, let code):
                    MotifCodeBlockView(
                        language: lang,
                        code: code,
                        tokens: tokens,
                        typo: typo,
                        onOpenInArtifactPane: { onOpenCodeInInspector(lang, code) }
                    )
                    .padding(.vertical, 2)

                case .table(_, let headers, let rows):
                    markdownTableView(headers: headers, rows: rows)
                }
            }
        }
    }

    private func groupContiguousProse(_ blocks: [MarkdownBlock]) -> [RenderedMarkdownSection] {
        var result: [RenderedMarkdownSection] = []
        var buffer: [MarkdownBlock] = []

        func flush() {
            guard !buffer.isEmpty else { return }
            let id = "prose_\(result.count)_\(buffer.first?.id ?? "")"
            result.append(.proseGroup(id, buffer))
            buffer.removeAll()
        }

        for block in blocks {
            switch block.kind {
            case .codeBlock(let lang, let code):
                flush()
                result.append(.codeBlock(block.id, lang, code))
            case .table(let headers, let rows):
                flush()
                result.append(.table(block.id, headers, rows))
            case .horizontalRule:
                flush()
                result.append(.horizontalRule(block.id))
            default:
                buffer.append(block)
            }
        }
        flush()
        return result
    }

    private func isListBlock(_ kind: MarkdownBlockKind) -> Bool {
        switch kind {
        case .bulletItem, .numberedItem: return true
        default: return false
        }
    }

    private func buildUnifiedAttributedString(for proseBlocks: [MarkdownBlock]) -> AttributedString {
        var full = AttributedString()
        let baseColor = overrideTextColor ?? tokens.primaryText
        let secondaryColor = overrideTextColor ?? tokens.secondaryText
        let mutedColor = overrideTextColor ?? tokens.mutedText

        for (index, block) in proseBlocks.enumerated() {
            switch block.kind {
            case .heading(let level, let text):
                let size: CGFloat = {
                    switch level {
                    case 1: return effectiveFontSize * 1.38
                    case 2: return effectiveFontSize * 1.24
                    case 3: return effectiveFontSize * 1.12
                    default: return effectiveFontSize * 1.04
                    }
                }()
                full.append(buildInlineAttributed(text, fontSize: size, isHeading: true, textColor: baseColor))

            case .bulletItem(let indent, let text):
                let indentSpaces = String(repeating: "   ", count: min(indent, 4))
                var bullet = AttributedString("\(indentSpaces)\(indent > 0 ? "◦" : "•")  ")
                bullet.font = typo.sans(size: effectiveFontSize, weight: .regular)
                bullet.foregroundColor = mutedColor
                full.append(bullet)
                full.append(buildInlineAttributed(text, fontSize: effectiveFontSize, isHeading: false, textColor: baseColor))

            case .numberedItem(let indent, let numStr, let text):
                let indentSpaces = String(repeating: "   ", count: min(indent, 4))
                var numAttr = AttributedString("\(indentSpaces)\(numStr) ")
                numAttr.font = typo.sans(size: max(11.5, effectiveFontSize - 0.5), weight: .semibold).monospacedDigit()
                numAttr.foregroundColor = secondaryColor
                full.append(numAttr)
                full.append(buildInlineAttributed(text, fontSize: effectiveFontSize, isHeading: false, textColor: baseColor))

            case .blockquote(let text):
                var bar = AttributedString("│  ")
                bar.font = typo.sans(size: effectiveFontSize, weight: .regular)
                bar.foregroundColor = mutedColor
                full.append(bar)
                full.append(buildInlineAttributed(text, fontSize: effectiveFontSize, isHeading: false, textColor: secondaryColor))

            case .paragraph(let text):
                full.append(buildInlineAttributed(text, fontSize: effectiveFontSize, isHeading: false, textColor: baseColor))

            default:
                break
            }

            if index < proseBlocks.count - 1 {
                let nextKind = proseBlocks[index + 1].kind
                let separator = (isListBlock(block.kind) && isListBlock(nextKind)) ? "\n" : "\n\n"
                full.append(AttributedString(separator))
            }
        }
        return full
    }

    private func buildInlineAttributed(_ raw: String, fontSize: CGFloat, isHeading: Bool, textColor: Color) -> AttributedString {
        var output = AttributedString()
        let parts = raw.components(separatedBy: "`")

        for (idx, part) in parts.enumerated() where !part.isEmpty {
            if idx % 2 == 1 {
                var codeAttr = AttributedString(" \(part) ")
                codeAttr.font = .system(size: max(11.0, fontSize - 1.5), weight: .regular, design: .monospaced)
                codeAttr.foregroundColor = textColor
                codeAttr.backgroundColor = tokens.codeSurface
                output.append(codeAttr)
            } else {
                let normalizedProse = LaTeXAndSymbolNormalizer.normalizeProse(part)
                var mdAttr = (try? AttributedString(markdown: normalizedProse, options: AttributedString.MarkdownParsingOptions(interpretedSyntax: .inlineOnlyPreservingWhitespace))) ?? AttributedString(normalizedProse)
                mdAttr.foregroundColor = textColor
                for run in mdAttr.runs {
                    let range = run.range
                    let isBold = run.inlinePresentationIntent?.contains(.stronglyEmphasized) == true || isHeading
                    let isItalic = run.inlinePresentationIntent?.contains(.emphasized) == true
                    var f = useMonospaceBody
                        ? Font.system(size: fontSize, weight: isBold ? .semibold : .regular, design: .monospaced)
                        : typo.sans(size: fontSize, weight: isBold ? .semibold : .regular)
                    if isItalic { f = f.italic() }
                    mdAttr[range].font = f
                }
                output.append(mdAttr)
            }
        }
        return output
    }

    @ViewBuilder
    private func markdownTableView(headers: [String], rows: [[String]]) -> some View {
        ScrollView(.horizontal, showsIndicators: false) {
            Grid(alignment: .leading, horizontalSpacing: 0, verticalSpacing: 0) {
                GridRow {
                    ForEach(Array(headers.enumerated()), id: \.offset) { _, header in
                        Text(header)
                            .font(typo.sans(size: max(12, effectiveFontSize - 1.0), weight: .semibold))
                            .foregroundColor(overrideTextColor ?? tokens.primaryText)
                            .textSelection(.enabled)
                            .padding(.horizontal, 12)
                            .padding(.vertical, 8)
                            .frame(maxWidth: .infinity, alignment: .leading)
                    }
                }
                .background(tokens.codeSurface)

                ForEach(Array(rows.enumerated()), id: \.offset) { rowIdx, row in
                    HairlineDivider(tokens: tokens)
                    GridRow {
                        ForEach(0..<headers.count, id: \.self) { colIdx in
                            let cell = colIdx < row.count ? row[colIdx] : ""
                            Text(LaTeXAndSymbolNormalizer.normalizeProse(cell))
                                .font(typo.sans(size: max(12, effectiveFontSize - 1.0)))
                                .foregroundColor(overrideTextColor ?? tokens.primaryText)
                                .textSelection(.enabled)
                                .padding(.horizontal, 12)
                                .padding(.vertical, 7)
                                .frame(maxWidth: .infinity, alignment: .leading)
                        }
                    }
                    .background(rowIdx % 2 == 1 ? tokens.codeSurface.opacity(0.45) : Color.clear)
                }
            }
            .clipShape(RoundedRectangle(cornerRadius: 8, style: .continuous))
            .overlay(
                RoundedRectangle(cornerRadius: 8, style: .continuous)
                    .strokeBorder(tokens.hairline, lineWidth: 1)
            )
        }
        .padding(.vertical, 3)
    }

    private func sanitizeIncompleteStreamMarkers(_ input: String, isStreaming: Bool) -> String {
        guard isStreaming else { return input }
        var result = input
        let boldCount = result.components(separatedBy: "**").count - 1
        if boldCount % 2 != 0, let lastRange = result.range(of: "**", options: .backwards) {
            result.removeSubrange(lastRange)
        }
        return result
    }

    private func matchHeading(_ trimmed: String) -> (level: Int, text: String)? {
        guard trimmed.hasPrefix("#") else { return nil }
        var count = 0
        for ch in trimmed { if ch == "#" { count += 1 } else { break } }
        guard count >= 1 && count <= 6 else { return nil }
        let remainder = String(trimmed.dropFirst(count))
        guard remainder.isEmpty || remainder.first == " " || remainder.first == "\t" else { return nil }
        var cleaned = remainder.trimmingCharacters(in: .whitespaces)
        while cleaned.hasSuffix("#") { cleaned.removeLast() }
        cleaned = cleaned.trimmingCharacters(in: .whitespaces)
        return cleaned.isEmpty ? nil : (count, cleaned)
    }

    private func splitTableRow(_ line: String) -> [String] {
        var trimmed = line.trimmingCharacters(in: .whitespaces)
        if trimmed.hasPrefix("|") { trimmed.removeFirst() }
        if trimmed.hasSuffix("|") { trimmed.removeLast() }
        return trimmed.components(separatedBy: "|").map { $0.trimmingCharacters(in: .whitespaces) }
    }

    private func isTableSeparatorLine(_ line: String) -> Bool {
        let trimmed = line.trimmingCharacters(in: .whitespaces)
        guard trimmed.contains("-") && trimmed.contains("|") else { return false }
        let allowed = CharacterSet(charactersIn: "|-: ")
        return trimmed.unicodeScalars.allSatisfy { allowed.contains($0) }
    }

    private func parseBlocks(from raw: String) -> [MarkdownBlock] {
        let lines = raw.components(separatedBy: .newlines)
        var blocks: [MarkdownBlock] = []
        var paragraphBuffer: [String] = []
        var idx = 0

        func flushParagraph() {
            let joined = paragraphBuffer.joined(separator: "\n").trimmingCharacters(in: .whitespacesAndNewlines)
            if !joined.isEmpty {
                blocks.append(MarkdownBlock(id: "p_\(blocks.count)", kind: .paragraph(joined)))
            }
            paragraphBuffer.removeAll()
        }

        while idx < lines.count {
            let line = lines[idx]
            let trimmed = line.trimmingCharacters(in: .whitespaces)

            if trimmed.hasPrefix("```") || trimmed.hasPrefix("~~~") {
                flushParagraph()
                let fenceChar = trimmed.first!
                let fenceLen = trimmed.prefix(while: { $0 == fenceChar }).count
                let lang = String(trimmed.dropFirst(fenceLen)).trimmingCharacters(in: .whitespaces)
                var codeLines: [String] = []
                idx += 1
                while idx < lines.count {
                    let candidate = lines[idx].trimmingCharacters(in: .whitespaces)
                    if candidate.prefix(while: { $0 == fenceChar }).count >= fenceLen
                        && candidate.drop(while: { $0 == fenceChar }).trimmingCharacters(in: .whitespaces).isEmpty {
                        idx += 1
                        break
                    }
                    codeLines.append(lines[idx])
                    idx += 1
                }
                blocks.append(MarkdownBlock(id: "code_\(blocks.count)", kind: .codeBlock(lang, codeLines.joined(separator: "\n"))))
                continue
            }

            if trimmed.contains("|"), idx + 1 < lines.count, isTableSeparatorLine(lines[idx + 1]) {
                flushParagraph()
                let headers = splitTableRow(trimmed)
                var rows: [[String]] = []
                idx += 2
                while idx < lines.count {
                    let rowLine = lines[idx].trimmingCharacters(in: .whitespaces)
                    if rowLine.isEmpty || !rowLine.contains("|") { break }
                    rows.append(splitTableRow(rowLine))
                    idx += 1
                }
                blocks.append(MarkdownBlock(id: "tbl_\(blocks.count)", kind: .table(headers, rows)))
                continue
            }

            let leadingSpaces = line.prefix(while: { $0 == " " || $0 == "\t" }).reduce(0) { $0 + ($1 == "\t" ? 4 : 1) }
            let indentLevel = leadingSpaces / 2

            if trimmed.isEmpty {
                flushParagraph()
            } else if trimmed == "---" || trimmed == "***" || trimmed == "___" {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "hr_\(blocks.count)", kind: .horizontalRule))
            } else if let heading = matchHeading(trimmed) {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "h_\(blocks.count)", kind: .heading(heading.level, heading.text)))
            } else if trimmed.hasPrefix("- ") || trimmed.hasPrefix("* ") || trimmed.hasPrefix("• ") || trimmed.hasPrefix("+ ") {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "b_\(blocks.count)", kind: .bulletItem(indentLevel, String(trimmed.dropFirst(2)))))
            } else if trimmed.hasPrefix("> ") || trimmed == ">" {
                flushParagraph()
                let quoteBody = trimmed.count > 2 ? String(trimmed.dropFirst(2)) : ""
                if !quoteBody.isEmpty {
                    blocks.append(MarkdownBlock(id: "q_\(blocks.count)", kind: .blockquote(quoteBody)))
                }
            } else if let num = matchNumberedList(trimmed) {
                flushParagraph()
                blocks.append(MarkdownBlock(id: "n_\(blocks.count)", kind: .numberedItem(indentLevel, num.number, num.rest)))
            } else {
                paragraphBuffer.append(line)
            }
            idx += 1
        }
        flushParagraph()
        return blocks
    }

    private func matchNumberedList(_ line: String) -> (number: String, rest: String)? {
        guard let sepIdx = line.firstIndex(of: ".") ?? line.firstIndex(of: ")") else { return nil }
        let prefix = String(line[..<sepIdx])
        guard !prefix.isEmpty, prefix.allSatisfy({ $0.isNumber }) else { return nil }
        let afterSep = line.index(after: sepIdx)
        guard afterSep < line.endIndex, line[afterSep] == " " else { return nil }
        return ("\(prefix).", String(line[line.index(after: afterSep)...]))
    }
}

public struct DraggableSplitHandle: View {
    @Binding public var width: CGFloat
    public let minWidth: CGFloat
    public let maxWidth: CGFloat
    public let invertDelta: Bool
    public let tokens: MotifTokens

    @State private var dragStartWidth: CGFloat? = nil
    @State private var isHovering: Bool = false

    public init(
        width: Binding<CGFloat>,
        minWidth: CGFloat,
        maxWidth: CGFloat,
        invertDelta: Bool = false,
        tokens: MotifTokens
    ) {
        self._width = width
        self.minWidth = minWidth
        self.maxWidth = maxWidth
        self.invertDelta = invertDelta
        self.tokens = tokens
    }

    public var body: some View {
        ZStack {
            Color.clear.frame(width: 7)
            Rectangle()
                .fill(isHovering ? tokens.strongBorder : tokens.hairline)
                .frame(width: isHovering ? 2 : 1)
        }
        .contentShape(Rectangle())
        .onHover { hovering in
            isHovering = hovering
            if hovering { NSCursor.resizeLeftRight.push() } else { NSCursor.pop() }
        }
        .gesture(
            DragGesture(minimumDistance: 1, coordinateSpace: .global)
                .onChanged { value in
                    if dragStartWidth == nil { dragStartWidth = width }
                    let base = dragStartWidth ?? width
                    let delta = invertDelta ? -value.translation.width : value.translation.width
                    width = min(maxWidth, max(minWidth, base + delta))
                }
                .onEnded { _ in dragStartWidth = nil }
        )
    }
}

// MARK: - Multi-Line Word-Wrapping Editor with Built-In Placeholder
public final class PlaceholderNSTextView: NSTextView {
    var placeholderString: String = ""
    var placeholderColor: NSColor = .secondaryLabelColor

    public override func draw(_ dirtyRect: NSRect) {
        super.draw(dirtyRect)
        if string.isEmpty && !hasMarkedText() && !placeholderString.isEmpty {
            let attrs: [NSAttributedString.Key: Any] = [
                .font: font ?? NSFont.systemFont(ofSize: 12.5),
                .foregroundColor: placeholderColor
            ]
            let inset = textContainerInset
            let pad = textContainer?.lineFragmentPadding ?? 5
            let rect = NSRect(
                x: inset.width + pad,
                y: inset.height,
                width: max(40, bounds.width - (inset.width + pad) * 2),
                height: max(20, bounds.height - inset.height * 2)
            )
            (placeholderString as NSString).draw(in: rect, withAttributes: attrs)
        }
    }
}

public struct MotifStyledTextEditor: NSViewRepresentable {
    @Binding public var text: String
    public let font: NSFont
    public let tokens: MotifTokens
    public var placeholder: String = ""

    public init(text: Binding<String>, font: NSFont, tokens: MotifTokens, placeholder: String = "") {
        self._text = text
        self.font = font
        self.tokens = tokens
        self.placeholder = placeholder
    }

    public func makeCoordinator() -> Coordinator { Coordinator(self) }

    public func makeNSView(context: Context) -> NSScrollView {
        let scrollView = NSScrollView()
        scrollView.drawsBackground = false
        scrollView.hasVerticalScroller = true
        scrollView.hasHorizontalScroller = false
        scrollView.autohidesScrollers = true
        scrollView.scrollerStyle = .overlay
        scrollView.borderType = .noBorder
        scrollView.appearance = NSAppearance(named: tokens.isEffectivelyDark ? .darkAqua : .aqua)

        let textView = PlaceholderNSTextView()
        textView.delegate = context.coordinator
        textView.drawsBackground = false
        textView.isRichText = false
        textView.allowsUndo = true
        textView.isAutomaticQuoteSubstitutionEnabled = false
        textView.isAutomaticDashSubstitutionEnabled = false
        textView.font = font
        textView.textColor = NSColor(tokens.primaryText)
        textView.insertionPointColor = NSColor(tokens.primaryText)
        textView.placeholderString = placeholder
        textView.placeholderColor = NSColor(tokens.mutedText)
        textView.textContainerInset = NSSize(width: 6, height: 8)
        textView.isVerticallyResizable = true
        textView.isHorizontallyResizable = false
        textView.autoresizingMask = [.width]
        textView.textContainer?.widthTracksTextView = true

        scrollView.documentView = textView
        return scrollView
    }

    public func updateNSView(_ scrollView: NSScrollView, context: Context) {
        context.coordinator.parent = self
        scrollView.scrollerStyle = .overlay
        scrollView.appearance = NSAppearance(named: tokens.isEffectivelyDark ? .darkAqua : .aqua)
        guard let textView = scrollView.documentView as? PlaceholderNSTextView else { return }
        if textView.string != text && !textView.hasMarkedText() {
            textView.string = text
            textView.needsDisplay = true
        }
        if textView.font != font { textView.font = font }
        let targetColor = NSColor(tokens.primaryText)
        if textView.textColor != targetColor {
            textView.textColor = targetColor
            textView.insertionPointColor = targetColor
        }
        let targetPhColor = NSColor(tokens.mutedText)
        if textView.placeholderString != placeholder || textView.placeholderColor != targetPhColor {
            textView.placeholderString = placeholder
            textView.placeholderColor = targetPhColor
            if textView.string.isEmpty { textView.needsDisplay = true }
        }
    }

    public final class Coordinator: NSObject, NSTextViewDelegate {
        var parent: MotifStyledTextEditor
        init(_ parent: MotifStyledTextEditor) { self.parent = parent }
        public func textDidChange(_ notification: Notification) {
            guard let tv = notification.object as? NSTextView, !tv.hasMarkedText() else { return }
            parent.text = tv.string
            tv.needsDisplay = true
        }
    }
}

        var parent: IMESafeComposerTextView
        var lastHandledFocusTrigger: Int
        private var lastMeasuredWidth: CGFloat = 0

        init(_ parent: IMESafeComposerTextView) {
            self.parent = parent
            self.lastHandledFocusTrigger = parent.focusTrigger

        public func textDidBeginEditing(_ notification: Notification) { parent.isFocused = true }
        public func textDidEndEditing(_ notification: Notification) { parent.isFocused = false }

        public func textDidChange(_ notification: Notification) {

            if !tv.hasMarkedText() { parent.text = tv.string }
            recalculateQuantizedMetrics(for: tv)

        public func handleWidthChangeIfNeeded(for textView: NSTextView) {

            guard abs(currentW - lastMeasuredWidth) > 6.0 else { return }
            lastMeasuredWidth = currentW
            recalculateQuantizedMetrics(for: textView)
        }

        public func recalculateQuantizedMetrics(for textView: NSTextView) {
            if textView.string.isEmpty {
                if parent.lineCount != 1 || parent.compactHeight != 32.0 {
                    DispatchQueue.main.async {
                        self.parent.lineCount = 1
                        self.parent.compactHeight = 32.0
                    }
                }
                return
            }
            guard let lm = textView.layoutManager, let tc = textView.textContainer else { return }
            lm.ensureLayout(for: tc)
            let usedH = lm.usedRect(for: tc).height
            let singleH = max(18.0, (textView.font?.ascender ?? 13.5) - (textView.font?.descender ?? -4.0) + (textView.font?.leading ?? 2.0))
            let lines = max(1, Int(round(usedH / singleH)))
            let targetH: CGFloat = lines <= 1 ? 32.0 : min(136.0, ceil(usedH) + 12.0)
            if parent.lineCount != lines || abs(parent.compactHeight - targetH) > 1.0 {
                DispatchQueue.main.async {
                    self.parent.lineCount = lines
                    self.parent.compactHeight = targetH
                }
            }
        }

        public func textView(_ textView: NSTextView, doCommandBy commandSelector: Selector) -> Bool {
            if textView.hasMarkedText() { return false }
            if parent.isSlashMenuVisible {
                if commandSelector == #selector(NSResponder.moveUp(_:)) { parent.onSlashMoveUp(); return true }
                if commandSelector == #selector(NSResponder.moveDown(_:)) { parent.onSlashMoveDown(); return true }
                if commandSelector == #selector(NSResponder.insertTab(_:)) {
                    let ok = parent.onSlashConfirm()
                    recalculateQuantizedMetrics(for: textView)
                    return ok
                }
                if commandSelector == #selector(NSResponder.cancelOperation(_:)) { parent.onSlashDismiss(); return true }
            }
            if commandSelector == #selector(NSResponder.insertNewline(_:)) {
                let flags = NSApp.currentEvent?.modifierFlags ?? []
                if flags.contains(.shift) {
                    textView.insertNewlineIgnoringFieldEditor(nil)
                    recalculateQuantizedMetrics(for: textView)
                    return true
                } else {
                    if parent.isSlashMenuVisible && parent.onSlashConfirm() {
                        recalculateQuantizedMetrics(for: textView)
                        return true
                    }
                    parent.onSend()
                    return true
                }
            }
            return false
        }
    }

    private func compactQuickSlider(
        label: String,
        value: Binding<Double?>,
        fallback: Double,
        range: ClosedRange<Double>
    ) -> some View {
        VStack(alignment: .leading, spacing: 4) {
            HStack(spacing: 4) {
                Text(label)
                    .font(typo.sans(size: 11.5, weight: .medium))
                    .foregroundColor(tokens.secondaryText)
                Spacer()
                if let v = value.wrappedValue {
                    Text(String(format: "%.2f", v))
                        .font(typo.sans(size: 11, weight: .semibold).monospacedDigit())
                        .foregroundColor(tokens.primaryText)
                    Button {
                        value.wrappedValue = nil
                        store.recordActiveModelParameterPreference()
                    } label: {
                        Image(systemName: "arrow.counterclockwise")
                            .font(.system(size: 9, weight: .medium))
                            .foregroundColor(tokens.mutedText)
                    }
                    .buttonStyle(.plain)
                    .help("Reset \(label) to model default")
                } else {
                    Text("Default")
                        .font(typo.sans(size: 11))
                        .foregroundColor(tokens.mutedText)
                }
            }
            Slider(
                value: Binding(
                    get: { value.wrappedValue ?? fallback },
                    set: {
                        value.wrappedValue = $0
                        store.recordActiveModelParameterPreference()
                    }
                ),
                in: range
            )
            .controlSize(.small)
            .tint(value.wrappedValue == nil ? tokens.mutedText.opacity(0.45) : tokens.primaryText)
        }
    }

    // MARK: - Responsive Desktop Conversation Stream (Session-Keyed So Chat Switching Never Goes Blank)
    private func conversationStreamCanvas(
        session: ConversationSession,
        columnWidth: CGFloat,
        horizontalPadding: CGFloat,
        opticalFontSize: CGFloat,
        isFullscreenOrWide: Bool
    ) -> some View {
        let thread = session.activeBranchThread
        let lastTurnID = thread.last?.id
        let dynamicBottomClearance = max(124.0, composerMeasuredHeight + 24.0)
        let jumpButtonBottomOffset = max(92.0, composerMeasuredHeight + 6.0)

        return ScrollViewReader { proxy in
            ScrollView(.vertical, showsIndicators: true) {
                LazyVStack(alignment: .leading, spacing: 26) {
                    ForEach(thread) { node in
                        ConversationTurnRow(
                            store: store,
                            node: node,
                            session: session,
                            isLatestTurn: node.id == lastTurnID,
                            opticalFontSize: opticalFontSize,
                            isFullscreenOrWide: isFullscreenOrWide,
                            columnWidth: columnWidth,
                            tokens: tokens,
                            editingMessageNodeID: $editingMessageNodeID,
                            editMessageDraft: $editMessageDraft,
                            clearThinkingOnAssistantEdit: $clearThinkingOnAssistantEdit,
                            isEditBoxExpandedFull: $isEditBoxExpandedFull,
                            confirmingDeleteNodeID: $confirmingDeleteNodeID
                        )
                        .id(node.id)
                    }

                    Color.clear
                        .frame(height: dynamicBottomClearance)
                        .id("STREAM_BOTTOM_ANCHOR")
                }
                .background(
                    ChatScrollBridgeView(
                        controller: scrollController,
                        isDark: tokens.isEffectivelyDark
                    ) { isAway in
                        store.isScrolledAwayFromBottom = isAway
                    }
                    .frame(width: 1, height: 1)
                )
                .frame(maxWidth: columnWidth, alignment: .leading)
                .padding(.horizontal, horizontalPadding)
                .padding(.top, 26)
                .frame(maxWidth: .infinity)
            }
            .id(session.id)
            .onAppear {
                DispatchQueue.main.async {
                    proxy.scrollTo("STREAM_BOTTOM_ANCHOR", anchor: .bottom)
                }
            }
            .onChange(of: store.streamScrollPulse) { _, _ in
                if store.isStreamingGeneration && !store.isScrolledAwayFromBottom && !store.isAutoScrollPaused {
                    scrollController.scrollToBottom(animated: false)
                }
            }
            .onChange(of: store.pendingScrollToMessageID) { _, targetID in
                if let id = targetID {
                    withAnimation(.spring(response: 0.28, dampingFraction: 0.86)) { proxy.scrollTo(id, anchor: .center) }
                    store.pendingScrollToMessageID = nil
                }
            }
            .overlay(alignment: .bottom) {
                HStack {
                    Spacer()
                    ScrollJumpOverlayView(
                        controller: scrollController,
                        isHidden: store.isComposerCanvasExpanded || !store.matchedSlashPersonas.isEmpty,
                        tokens: tokens,
                        onJumpTop: {
                            store.isAutoScrollPaused = true
                            scrollController.scrollToTop(animated: true)
                        },
                        onJumpBottom: {
                            store.isScrolledAwayFromBottom = false
                            store.isAutoScrollPaused = false
                            scrollController.scrollToBottom(animated: true)
                        }
                    )
                }
                .frame(maxWidth: columnWidth)
                .padding(.horizontal, horizontalPadding)
                .padding(.bottom, jumpButtonBottomOffset)
            }
        }
    }
}

// MARK: - Single Conversation Turn (Rock-Solid Equatable Prose + Dual Top/Bottom Thinking Collapse)
private struct ConversationTurnRow: View {
    @ObservedObject var store: WorkstationStore
    let node: ConversationMessageNode
    let session: ConversationSession
    let isLatestTurn: Bool
    let opticalFontSize: CGFloat
    let isFullscreenOrWide: Bool
    let columnWidth: CGFloat
    let tokens: MotifTokens

    @Binding var editingMessageNodeID: UUID?
    @Binding var editMessageDraft: String
    @Binding var clearThinkingOnAssistantEdit: Bool
    @Binding var isEditBoxExpandedFull: Bool
    @Binding var confirmingDeleteNodeID: UUID?

    @State private var isHovered: Bool = false
    @State private var isCopied: Bool = false
    @State private var isBottomCollapseHovered: Bool = false

    private var typo: TypographyConfiguration { store.typography }
    private var isNodeStreaming: Bool { store.isStreamingGeneration && store.streamingNodeID == node.id }
    private var siblingMeta: (currentIndex: Int, totalCount: Int) { session.siblingInfo(for: node.id) }

    var body: some View {
        VStack(alignment: node.role == .user && editingMessageNodeID != node.id ? .trailing : .leading, spacing: 8) {
            if editingMessageNodeID == node.id {
                inlineMessageEditor
            } else if node.role == .user {
                userTurnBody
            } else {
                assistantTurnBody
            }

            if let err = node.errorNote {
                Text(err)
                    .font(typo.sans(size: 12.5))
                    .foregroundColor(.red)
                    .frame(maxWidth: .infinity, alignment: .leading)
            }

            if editingMessageNodeID != node.id {
                turnActionAndTelemetryBar
            }
        }
        .frame(maxWidth: .infinity, alignment: node.role == .user && editingMessageNodeID != node.id ? .trailing : .leading)
        .onHover { hovering in
            withAnimation(.easeOut(duration: 0.14)) { isHovered = hovering }
        }
        .contextMenu {
            Button("Copy Message") { copyNodeText() }
            Button(node.role == .user ? "Edit Prompt…" : "Edit / Trim Reply to Save KV RAM…") { beginEditing() }
            Button("Branch to New Chat from Here") { store.branchMessageToNewChat(upToNodeID: node.id) }
            Divider()
            Button("Delete Message (Free ~\(node.estimatedTokenFootprint) tok)…", role: .destructive) {
                confirmingDeleteNodeID = node.id
            }
        }
    }

    // MARK: - User Turn (Right-Aligned Soft Neutral Pill)
    private var userTurnBody: some View {
        VStack(alignment: .trailing, spacing: 8) {
            if !node.attachments.isEmpty {
                HStack(spacing: 6) {
                    ForEach(node.attachments) { att in
                        HStack(spacing: 5) {
                            Image(systemName: att.kind == .downscaledVisionImage ? "photo" : "doc.text")
                                .font(.system(size: 10))
                                .foregroundColor(tokens.secondaryText)
                            Text("\(att.title) · \(att.formattedByteSize)")
                                .font(typo.sans(size: 11, weight: .regular).monospacedDigit())
                                .foregroundColor(tokens.secondaryText)
                        }
                        .padding(.horizontal, 9).padding(.vertical, 4.5)
                        .background(tokens.codeSurface)
                        .clipShape(RoundedRectangle(cornerRadius: 8, style: .continuous))
                        .overlay(RoundedRectangle(cornerRadius: 8, style: .continuous).strokeBorder(tokens.hairline, lineWidth: 1))
                    }
                }
            }

            if !node.content.isEmpty {
                Text(node.content)
                    .font(typo.sans(size: opticalFontSize, weight: .regular))
                    .lineSpacing(opticalFontSize * 0.42)
                    .foregroundColor(tokens.primaryText)
                    .textSelection(.enabled)
                    .fixedSize(horizontal: false, vertical: true)
                    .padding(.horizontal, 16)
                    .padding(.vertical, 11)
                    .background(tokens.userBubbleSurface)
                    .clipShape(RoundedRectangle(cornerRadius: 16, style: .continuous))
                    .frame(maxWidth: max(280, columnWidth * 0.78), alignment: .trailing)
            }
        }
        .frame(maxWidth: .infinity, alignment: .trailing)
    }

    // MARK: - Assistant Turn (Distinct Indented Reasoning Marginalia + Full-Width Primary Answer)
    private var assistantTurnBody: some View {
        VStack(alignment: .leading, spacing: 14) {
            if !node.thinkingTrace.isEmpty || (!node.isThinkingComplete && isNodeStreaming) {
                editorialThinkingSection
            }

            if node.content.isEmpty && isNodeStreaming {
                if store.isLoadingModelWeights {
                    let isHoldingDone = store.isHoldingLoadCompletionState
                    let frac = isHoldingDone ? 1.0 : min(0.99, max(0.02, store.memorySnapshot.loadProgressAgainstTarget))
                    VStack(alignment: .leading, spacing: 6) {
                        Text(isHoldingDone
                             ? "\(store.loadingModelName) loaded into Unified Memory (100%)"
                             : "Loading \(store.loadingModelName) into Unified Memory (\(Int(frac * 100))%)…")
                            .font(typo.sans(size: 12.5, weight: .regular).monospacedDigit())
                            .foregroundColor(isHoldingDone ? tokens.statusOnline : tokens.mutedText)
                        ProgressView(value: frac)
                            .tint(isHoldingDone ? tokens.statusOnline : tokens.primaryText)
                            .frame(maxWidth: 240)
                    }
                } else if store.isEvaluatingConversationContext && node.thinkingTrace.isEmpty {
                    Text("Reading conversation context…")
                        .font(typo.sans(size: 13))
                        .foregroundColor(tokens.mutedText)
                } else if node.thinkingTrace.isEmpty {

    private func buildTelemetryFooter() -> (text: String, isWarning: Bool) {
        let answerTok = node.content.isEmpty ? 0 : max(node.estimatedTokenFootprint, node.metrics?.totalTokens ?? 0)
        let m = node.metrics
        let tps = m?.tokensPerSecond ?? 0.0
        let cached = m?.promptEvalCachedTokens ?? 0
        let promptTok = m?.promptEvalTokens ?? 0
        let doneReason = m?.doneReason ?? ""

        var parts: [String] = []
        if tps > 0 {
            parts.append(String(format: "%.1f tok/s", tps))
            parts.append("\(answerTok) tok")
        } else {
            parts.append("~\(answerTok) tok")
        }
        if promptTok > 0 && cached > 0 {
            let pct = Int((Double(cached) / Double(promptTok)) * 100.0)
            parts.append("KV \(pct)%")
        }
        if doneReason == "length" { parts.append("Truncated") }
        return (parts.joined(separator: " · "), doneReason == "length")
